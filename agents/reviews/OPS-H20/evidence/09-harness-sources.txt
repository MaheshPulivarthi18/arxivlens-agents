"""AC1/AC4: recompute the job/step arithmetic from the two cited jobs' REST step timings and log figures."""
import datetime as dt, re, sys
def steps(path):
    rows = {}
    for line in open(path):
        n, name, concl, s, e = line.rstrip("\n").split("|")
        p = lambda t: dt.datetime.strptime(t, "%Y-%m-%dT%H:%M:%SZ")
        rows[int(n)] = (name, concl, (p(e) - p(s)).total_seconds(), p(s), p(e))
    return rows
def logv(path, pat):
    for line in open(path, encoding="utf-8", errors="replace"):
        m = re.search(pat, line)
        if m: return line[:28], m.group(1)
    return None, None
def ts(path, pat):
    for line in open(path, encoding="utf-8", errors="replace"):
        if re.search(pat, line):
            return dt.datetime.strptime(line[:26], "%Y-%m-%dT%H:%M:%S.%f")
mm = lambda s: f"{int(s // 60)}m{int(s % 60):02d}s ({s / 60:.1f} min)"
J = {"main 44764887789 job 136386731578": "136386731578", "PR#111 run 44899068259 job 136830894337": "136830894337"}
res = {}
for label, j in J.items():
    st, log = steps(f"steps-{j}.txt"), f"log-{j}.txt"
    cpu = logv(log, r"Model name:\s+(.*)")[1].strip()
    rr = float(logv(log, r"Hybrid RRF \+ Rerank\s+\|.*\|\s*([0-9.]+)\s*$")[1])
    t0 = ts(log, r"hardware: embedder device=")
    t1 = ts(log, r"Benchmark Results Summary Table")
    pre = logv(log, r"\[replay\] prefetch: (.*)")[1]
    tout = logv(log, r"(##\[error\]The action 'make eval' has timed out.*)")[1]
    gate = logv(log, r"(GATE: \w+)")[1]
    setup = sum(st[n][2] for n in range(1, 11))
    res[label] = dict(rr=rr, retr=(t1 - t0).total_seconds(), step=st[19][2])
    print(f"== {label}\n  CPU: {cpu}")
    print(f"  setup+stack-up (steps 1-10): {mm(setup)}   prefetch (12): {mm(st[12][2])}  {pre}")
    print(f"  self-test (16): {mm(st[16][2])}   eval-seed (17): {mm(st[17][2])}")
    print(f"  make eval (19): {st[19][1]} {mm(st[19][2])}   retrieval phase (log): {mm((t1 - t0).total_seconds())}")
    print(f"  rerank avg ms/query: {rr}   GATE line: {gate}   timeout line: {tout}")
    print(f"  upload/teardown (20-23): {mm(sum(st[n][2] for n in range(20, 24)))}   job total: {mm((st[47][4] - st[1][3]).total_seconds())}")
m, p = res[list(J)[0]], res[list(J)[1]]

"""Observation: worst-case wall time of the prefetch step from scripts/arxiv_replay.py's own constants (PR head)."""
import ast, sys
src = open(sys.argv[1]).read()
c = {n.targets[0].id: ast.literal_eval(n.value) for n in ast.parse(src).body
     if isinstance(n, ast.Assign) and isinstance(n.targets[0], ast.Name) and n.targets[0].id in
     ("INTERVAL", "RETRY_WAITS", "MAX_WAIT", "TIMEOUT")}
print("constants:", c)
attempts = len(c["RETRY_WAITS"]) + 1
per_url = attempts * (c["INTERVAL"] + c["TIMEOUT"]) + len(c["RETRY_WAITS"]) * c["MAX_WAIT"]
print(f"one URL, every attempt hangs to TIMEOUT, Retry-After >= MAX_WAIT: {attempts} x ({c['INTERVAL']} + {c['TIMEOUT']}) + "
      f"{len(c['RETRY_WAITS'])} x {c['MAX_WAIT']} = {per_url:.0f} s")
per_query = 2 * per_url  # prefetch_query: the id_list API feed + its PDF (1 entry per id_list query)
print(f"one id_list query (API feed + 1 PDF): {per_query:.0f} s")
budget = 2520
print(f"budget is checked only before each query, so the last query can start at ~{budget} s and end at "
      f"~{budget + per_query:.0f} s = {(budget + per_query) / 60:.1f} min")
print(f"the job comment counts 2520 + {c['TIMEOUT']} = {(budget + c['TIMEOUT']) / 60:.0f} min; the prefetch step's own cap is 45 min, "
      f"so a job that reaches 'make eval' spent <= 45 min there: 10 + 45 + 1 + 70 + 45 + 2 = {10 + 45 + 1 + 70 + 45 + 2} <= 180")

"""AC1/AC2: PyYAML diff of every timeout-minutes in eval.yml, main vs PR head, plus a structural equality check."""
import copy, subprocess, sys, yaml
base, head, path = sys.argv[1], sys.argv[2], ".github/workflows/eval.yml"
def load(ref):
    return yaml.safe_load(subprocess.run(["git", "show", f"{ref}:{path}"], capture_output=True, text=True, check=True).stdout)
def timeouts(doc):
    out = []
    for jn, job in doc["jobs"].items():
        out.append((f"job {jn}", job.get("timeout-minutes")))
        for i, st in enumerate(job.get("steps", []), 1):
            out.append((f"step {i:2d} {st.get('name') or st.get('uses')}"[:78], st.get("timeout-minutes")))
    return out
a, b = load(base), load(head)
ta, tb = timeouts(a), timeouts(b)
assert [k for k, _ in ta] == [k for k, _ in tb], "step list differs"
print(f"{'':80} main  head")
changed = 0
for (k, va), (_, vb) in zip(ta, tb):
    flag = "  <-- CHANGED" if va != vb else ""
    changed += va != vb
    print(f"{k:80} {str(va):>4}  {str(vb):>4}{flag}")
print(f"\nentries: {len(ta)}  changed: {changed}")
def strip(doc):
    d = copy.deepcopy(doc)
    for job in d["jobs"].values():
        job.pop("timeout-minutes", None)
        for st in job.get("steps", []):
            st.pop("timeout-minutes", None)
    return d
print("parsed YAML identical apart from timeout-minutes:", strip(a) == strip(b))
st = lambda d: [s for s in d["jobs"]["eval"]["steps"] if s.get("id") == "prefetch"][0]
print("prefetch run unchanged:", st(a)["run"] == st(b)["run"], "|", st(b)["run"].strip())

print(f"sum of all step timeouts at head (if every step hit its cap): {sum_head} min")
