// Minimal stand-in for the vitest API used by src/test/tokens.test.js (describe/it/it.each/expect). Logic check only; CI's Vitest log is the pass/fail.
const fmt = (v) => JSON.stringify(v);
let pass = 0, fail = 0; const log = [];
const matchers = (v, neg) => {
  const ok = (c, msg) => { if (c !== neg) return; throw new Error(`${neg ? 'not.' : ''}${msg}`); };
  return {
    toMatch: (re) => ok(re.test(v), `toMatch ${re} on ${fmt(v)}`),
    toEqual: (x) => ok(fmt(v) === fmt(x), `toEqual: got ${fmt(v)}`),
    toBeGreaterThan: (x) => ok(v > x, `toBeGreaterThan ${x}: got ${v}`),
  };
};
export const expect = (v) => ({ ...matchers(v, false), not: matchers(v, true) });
const run = (name, fn) => { try { fn(); pass++; log.push(`  ok   ${name}`); } catch (e) { fail++; log.push(`  FAIL ${name}\n         ${e.message}`); } };
export const it = (name, fn) => run(name, fn);
it.each = (rows) => (name, fn) => rows.forEach((r) => run(name.replace('%s', r), () => fn(r)));
export const describe = (name, fn) => { log.push(name); fn(); };
process.on('exit', () => { console.log(log.join('\n')); console.log(`\n${pass} passed, ${fail} failed`); process.exitCode = fail ? 1 : 0; });
