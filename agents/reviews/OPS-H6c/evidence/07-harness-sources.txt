package kafka

// QA probe (not in the PR): fake fidelity of deadlinePoll vs the real franz-go v1.16.1 client on a deadline ctx.
import (
	"bytes"
	"context"
	"log"
	"strings"
	"testing"
	"time"

	"github.com/twmb/franz-go/pkg/kgo"
)

func TestQARealKgoDeadline(t *testing.T) {
	cl, err := kgo.NewClient(kgo.SeedBrokers("127.0.0.1:1"), kgo.ConsumerGroup("qa"), kgo.ConsumeTopics(topic))
	if err != nil {
		t.Fatal(err)
	}
	defer cl.Close()
	ctx, cancel := context.WithTimeout(context.Background(), 20*time.Millisecond)
	defer cancel()
	fs := cl.PollFetches(ctx)
	fs.EachError(func(tp string, p int32, e error) {
		t.Logf("real kgo PollFetches on deadline ctx: topic=%q partition=%d err=%v (err==ctx.Err(): %v)", tp, p, e, e == ctx.Err())
	})

	var buf bytes.Buffer
	defer log.SetOutput(log.Writer())
	log.SetOutput(&buf)
	cl2, _ := kgo.NewClient(kgo.SeedBrokers("127.0.0.1:1"), kgo.ConsumerGroup("qa"), kgo.ConsumeTopics(topic))
	defer cl2.Close()
	c := &Consumer{client: cl2, group: "qa", topics: []string{topic}, maxRetries: 3}
	ctx2, cancel2 := context.WithTimeout(context.Background(), 20*time.Millisecond)
	defer cancel2()
	start := time.Now()
	rerr := c.Start(ctx2, func(context.Context, *kgo.Record) error { return nil })
	out := buf.String()
	log.SetOutput(log.Writer())
	t.Logf("Start(real kgo, 20ms deadline) = %v in %v; Poll error lines: %d; shutdown line: %v", rerr, time.Since(start).Round(time.Millisecond),
		strings.Count(out, "Poll error"), strings.Contains(out, "Context cancelled, shutting down"))
	for _, l := range strings.Split(strings.TrimSpace(out), "\n") {
		t.Logf("  log: %s", l)
	}
}

#!/usr/bin/env python3
"""Single-edit mutants of internal/kafka/consumer.go (PR head 6b1e814) in the real franz-go v1.16.1 harness.
Each mutant: fresh mktemp -d copy of the harness (never deleted), one str.replace that must match exactly once
(anchored on the code line, not comments), then go test -race -count=1 twice: full package and
-run '^TestConsumerPollErrorAfterDeadline$'. Build failure -> INVALID (never KILLED). Prints the mutant diff line."""
import os, re, shutil, subprocess, sys, tempfile
H = os.path.join(os.path.dirname(os.path.abspath(__file__)), "..", "harness")
F = "internal/kafka/consumer.go"
GL = "\t\t\tif ctx.Err() != nil && errors.Is(err, ctx.Err()) {\n"
G = GL + "\t\t\t\treturn\n\t\t\t}\n"
M = [
 ("CONTROL (no edit)", None, None),
 ("M9 (AC3a) errors.Is(err, ctx.Err()) -> errors.Is(err, context.Canceled)", GL, GL.replace("errors.Is(err, ctx.Err())", "errors.Is(err, context.Canceled)")),
 ("AC3b remove the guard", G, ""),
 ("X1 guard only on DeadlineExceeded (cancel rows should catch)", GL, GL.replace("errors.Is(err, ctx.Err())", "errors.Is(err, context.DeadlineExceeded)")),
 ("X2 ctx.Err() == context.Canceled && errors.Is(err, ctx.Err())", GL, GL.replace("ctx.Err() != nil &&", "ctx.Err() == context.Canceled &&")),
 ("X3 err == context.Canceled (identity, not errors.Is)", GL, "\t\t\tif ctx.Err() != nil && err == context.Canceled {\n"),
]
os.environ.update(GOTOOLCHAIN="local", GOFLAGS="-mod=mod", GOPROXY="off", GOSUMDB="off")
def run(d, extra):
    p = subprocess.run(["go","test","-race","-count=1",*extra,"./internal/kafka/"], cwd=d, capture_output=True, text=True)
    out = p.stdout + p.stderr
    if p.returncode == 0: return "PASS (survived)" , []
    if re.search(r"\[build failed\]|cannot use|undefined:|declared and not used|syntax error", out): return "INVALID (build)", out.splitlines()[:5]
    fails = sorted(set(re.findall(r"--- FAIL: (\S+)", out)))
    if not fails: return "INVALID (no --- FAIL line)", out.splitlines()[:5]
    return "FAIL (killed)", fails
for name, old, new in M:
    d = tempfile.mkdtemp(prefix="mut-", dir=os.path.dirname(H))  # sibling of harness so ../<dep> replaces resolve
    shutil.copytree(H, d, dirs_exist_ok=True)
    src = open(os.path.join(d, F)).read()
    if old is not None:
        n = src.count(old)
        if n != 1: print(f"{name}: SKIP, anchor matched {n}x"); continue
        src = src.replace(old, new); open(os.path.join(d, F), "w").write(src)
    diff = subprocess.run(["diff", os.path.join(H, F), os.path.join(d, F)], capture_output=True, text=True).stdout.strip()
    full, ff = run(d, [])
    one, of = run(d, ["-run", "^TestConsumerPollErrorAfterDeadline$"])
    print(f"== {name}")
    print("   diff: " + (" | ".join(l for l in diff.splitlines() if l[:1] in "<>") or "(none)"))
    print(f"   full package: {full} {' '.join(ff)}")
    print(f"   -run ^TestConsumerPollErrorAfterDeadline$: {one} {' '.join(of)}")

done
