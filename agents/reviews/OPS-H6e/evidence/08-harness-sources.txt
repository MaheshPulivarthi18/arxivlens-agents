"""QA OPS-H6e SQL probe: runs the save_chunk_embeddings statements of a given app tree against a real local PostgreSQL 16
(one fresh DB per run), single-session (AC2-AC4) and two-session (concurrent processor UPDATE / tail DELETE).
pgvector can't be built in this sandbox, so `halfvec` is a text DOMAIN; the vector type is not what is under test.
Statements are taken from the tree's real services/common/db.py by calling save_chunk_embeddings with a recording cursor.
usage: sql_probe.py <app_tree> <pg_socket_dir> <label>"""

class Rec:
    def __init__(self): self.calls, self.rowcount = [], 1
    def __enter__(self): return self
    def __exit__(self, *a): return False
    def execute(self, sql, params=()): self.calls.append((sql, params))
class RecConn:
    def __init__(self): self.c = Rec()
    def cursor(self, **k): return self.c
    def commit(self): pass

def lit(v):
    if isinstance(v, int): return str(v)
    return "'" + str(v).replace("'", "''") + "'"
def render(item):
    conn = RecConn()
    try:
        db.save_chunk_embeddings(conn, [item])
    except Exception as e:  # noqa: BLE001
        return f"-- save_chunk_embeddings raised {type(e).__name__}: {e}"
    out = []
    for sql, params in conn.c.calls:
        s = sql
        for p in params: s = s.replace("%s", lit(p), 1)
        out.append(" ".join(s.split()))
    return "\n".join(out)
