===== standin.py =====
"""QA stand-in modules (NOT the real BGE tokenizer): a whitespace word-level 'fast' tokenizer with offset mapping,
plus MagicMock stubs for pymupdf/torch/transformers.AutoModel and the processor worker module (PDF tests can't run)."""
import re, sys, types
from unittest import mock

class StandInFastTokenizer:
    is_fast = True
    def __init__(self):
        self.vocab, self.inv = {}, {}
        self.model_max_length = 512
    def _id(self, w):
        if w not in self.vocab:
            i = len(self.vocab) + 1000
            self.vocab[w], self.inv[i] = i, w
        return self.vocab[w]
    def _spans(self, text):
        return [(m.group(0).lower(), m.start(), m.end()) for m in re.finditer(r"\S+", text)]
    def tokenize(self, text):
        return [w for w, _, _ in self._spans(text)]
    def encode(self, text, add_special_tokens=True):
        return [self._id(w) for w, _, _ in self._spans(text)]
    def __call__(self, text, add_special_tokens=True, return_offsets_mapping=False):
        sp = self._spans(text)
        out = {"input_ids": [self._id(w) for w, _, _ in sp]}
        if return_offsets_mapping:
            out["offset_mapping"] = [(s, e) for _, s, e in sp]
        return out
    def decode(self, ids, skip_special_tokens=True):
        return " ".join(self.inv[i] for i in ids)

def install():
    tr = types.ModuleType("transformers")
    tr.AutoTokenizer = types.SimpleNamespace(from_pretrained=lambda name: StandInFastTokenizer())
    tr.AutoModel = mock.MagicMock()
    sys.modules["transformers"] = tr
    for m in ("pymupdf", "torch"):
        sys.modules[m] = mock.MagicMock()
    w = types.ModuleType("services.processor.worker")
    w.ProcessorWorker = mock.MagicMock()
    sys.modules["services.processor.worker"] = w

===== probe.py =====
"""QA probe (stand-in tokenizer). argv[1] = head m0_skeleton.py, argv[2] = base (origin/main) m0_skeleton.py.
P1 byte identity: for random sections (random lead ws, random breaks), head chunks minus page_start == base chunks.
P2 oracle: page_start == page of the last break <= the char where the chunk body occurs in the unique raw text
   (found with str.find, not offsets); key absent iff that char < first break. P3 unknown cases never get the key."""
import importlib.util, os, random, sys
sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
import standin; standin.install()

def load(path, name):
    spec = importlib.util.spec_from_file_location(name, path); m = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(m); return m

head, base = load(sys.argv[1], "head_m0"), load(sys.argv[2], "base_m0")
WORDS = [f"w{i}" for i in range(400)]
fails = {"P1": 0, "P2": 0, "P3": 0}; n_chunks = n_paged = n_unpaged = 0; first = []

def mk(cls, rng, tok):
    c = cls.TokenChunker("stand-in"); c.tokenizer = tok
    c.target_tokens, c.overlap_tokens, c.min_tokens = rng.choice([(60, 20, 5), (40, 10, 3), (120, 30, 8), (512, 75, 50)])
    return c

for trial in range(400):
    rng = random.Random(trial)
    tok = standin.StandInFastTokenizer()
    h, b = mk(head, rng, tok), mk(base, random.Random(trial), tok)
    secs = []
    for s in range(rng.randint(1, 3)):
        n = rng.randint(30, 700)
        lead = rng.choice(["", " ", "\n\t ", "   "])
        body = " ".join(rng.sample(WORDS, min(n, 400)) + [f"x{trial}_{s}_{i}" for i in range(max(0, n - 400))])
        raw = lead + body + rng.choice(["", " ", "\n"])
        starts = [i for i, ch in enumerate(raw) if ch != " " and (i == 0 or raw[i - 1] == " ")]
        kind = rng.choice(["breaks", "breaks", "breaks", "none", "empty", "late"])
        sec = {"title": f"Sec {s}", "text": raw, "page": rng.randint(1, 5), "level": 1}
        if kind in ("breaks", "late"):
            k = rng.randint(1, 5)
            offs = sorted(set(rng.sample(range(len(raw)), k)))  # any char, word starts or not
            if kind == "breaks" and rng.random() < 0.7:
                offs = sorted(set([0] + offs))
            p0 = sec["page"]; sec["page_breaks"] = [(o, p0 + j) for j, o in enumerate(offs)]
        elif kind == "empty":
            sec["page_breaks"] = []
        secs.append(sec)
    plain = [{k: v for k, v in s.items() if k != "page_breaks"} for s in secs]
    got, want = h.chunk_sections("Title X", secs), b.chunk_sections("Title X", plain)
    if [{k: v for k, v in c.items() if k != "page_start"} for c in got] != want:
        fails["P1"] += 1; first.append(("P1", trial))
    by_title = {s["title"]: s for s in secs}
    for c in got:
        n_chunks += 1
        s = by_title[c["section_title"]]; pb = s.get("page_breaks") or []
        if not pb:
            if "page_start" in c: fails["P3"] += 1; first.append(("P3", trial))
            continue
        bodytxt = c["content"].split("\n\n", 1)[1]
        raw = s["text"]; assert raw.count(bodytxt) == 1
        pos = raw.find(bodytxt)
        pages = [p for o, p in pb if o <= pos]
        exp = pages[-1] if pages else None
        if c.get("page_start") != exp or (exp is None and "page_start" in c):
            fails["P2"] += 1; first.append(("P2", trial, c["chunk_index"], c.get("page_start"), exp))
        if "page_start" in c: n_paged += 1
        else: n_unpaged += 1
print(f"RESULT trials=400 chunks={n_chunks} with_page_start={n_paged} breaks_but_unknown={n_unpaged} "
      f"P1_identity_fail={fails['P1']} P2_oracle_fail={fails['P2']} P3_unknown_fail={fails['P3']}")
print("first failures:", first[:5])
sys.exit(1 if any(fails.values()) else 0)
