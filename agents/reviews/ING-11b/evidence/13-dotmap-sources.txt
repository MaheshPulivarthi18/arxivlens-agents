"""Collect the stack.yml `make test-full` pytest set offline: any module that fails to import (psycopg, torch, kafka...)
is replaced by a MagicMock module, so pytest can list node IDs in run order. Usage: collect_with_mocks.py <args...>"""
import importlib.abc, importlib.machinery, sys, types
from unittest import mock
class _Mock(types.ModuleType):
    def __getattr__(self, n):
        if n.startswith('__'): raise AttributeError(n)
        return mock.MagicMock(name=f"{self.__name__}.{n}")
class Finder(importlib.abc.MetaPathFinder, importlib.abc.Loader):
    SKIP = {"simplejson"}  # optional deps that libraries probe with try/except: leave them missing
    def find_spec(self, name, path, target=None):
        if name.split(".")[0] in self.SKIP: return None
        for f in sys.meta_path:
            if f is self: continue
            try:
                s = f.find_spec(name, path, target)
            except Exception:
                s = None
            if s: return None
        return importlib.machinery.ModuleSpec(name, self, is_package=True)
    def create_module(self, spec): m = _Mock(spec.name); m.__path__ = []; return m
    def exec_module(self, m): pass
sys.meta_path.append(Finder())
import pytest
sys.exit(pytest.main(sys.argv[1:]))

"""Map the stack.yml `make test-full` progress dots to node IDs. ids.txt = offline collection (collect_with_mocks.py) of the
same pytest args at the PR head; the log's dot lines are concatenated in order; the char at position i is test i's outcome."""
import re, sys
ids = [l.strip() for l in open(sys.argv[1]) if '::' in l and not l.startswith(('ERROR', 'FAILED'))]
log = open(sys.argv[2]).read().splitlines()
start = next(i for i, l in enumerate(log) if 'python3 -m pytest -q tests/ --deselect' in l)
dots = ''
for l in log[start + 1:]:
    m = re.match(r'^\S+Z ([.sFEx]+)\s+\[\s*\d+%\]$', l)
    if not m: break
    dots += m.group(1)
print(f"collected offline: {len(ids)} ids; outcome chars in log: {len(dots)} (dots={dots.count('.')}, s={dots.count('s')}, F={dots.count('F')}, E={dots.count('E')})")
for want in ['test_evals_data.py::test_chunk_counts_sql_against_real_postgres', 'test_m2_pipeline.py::']:
