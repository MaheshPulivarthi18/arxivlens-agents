#!/usr/bin/env python3
"""Round 2 extra mutants (X1-X5) aimed at the new r2 test rows. Single-edit mutants of internal/dlq/manager.go (PR #83 head) in the real-franz-go go1.24 harness.
Each mutant: fresh copy of the harness (mktemp -d), one str.replace that must match exactly once,
then `go test -count=1 ./internal/dlq/` (unit tests only, no integration tag).
Build/vet failure -> INVALID (never KILLED). Prints the mutant's changed line next to the verdict."""
import os, re, shutil, subprocess, sys, tempfile
S = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
H = os.environ.get("HARNESS", os.path.join(S, "harness")); F = "internal/dlq/manager.go"
M = [
 ("X1 nextCursor: next partition chosen by High > 0, not Low < High (lands on empty partition with Low=High>0)", "\t\tif wm[p].Low < wm[p].High {\n\t\t\treturn Cursor{Partition: p,", "\t\tif wm[p].High > 0 {\n\t\t\treturn Cursor{Partition: p,"),
 ("X2 From at/after High clamped to High-1 (re-reads the last record)", "\t\t\t\tfrom = opt.From.Offset\n", "\t\t\t\tfrom = min(opt.From.Offset, wm[p].High-1)\n"),
 ("X3 false expiry warning when From is at/after High", "\t\t\tif from > opt.From.Offset {", "\t\t\tif from > opt.From.Offset || opt.From.Offset >= wm[p].High {"),
 ("X4 empty --partition page reports a Next cursor at Low", "\treturn page, nil\n}\n\n// listPartitions", "\tif opt.Partition != nil && len(page.Records) == 0 {\n\t\tpage.Next = Cursor{Partition: *opt.Partition, Offset: wm[*opt.Partition].Low, set: true}\n\t}\n\treturn page, nil\n}\n\n// listPartitions"),
 ("X5 page-full check > instead of >= (a full page never sets Next mid-topic)", "\t\tif len(page.Records) >= size {\n\t\t\tpage.Next", "\t\tif len(page.Records) > size {\n\t\t\tpage.Next"),
]
src = open(os.path.join(H, F)).read()
env = dict(os.environ, GOTOOLCHAIN="local", GOFLAGS="-mod=mod", GOPROXY="off")
rows = []
for name, old, new in M:
    n = src.count(old)
    if n != 1:
        rows.append((name, "BADPATTERN(%d)" % n, "", "")); continue
    d = tempfile.mkdtemp(prefix="mut-", dir=os.path.join(S, "runs"))
    shutil.copytree(H, os.path.join(d, "h")); w = os.path.join(d, "h")
    open(os.path.join(w, F), "w").write(src.replace(old, new))
    changed = " | ".join(l.strip() for l in new.splitlines() if l.strip() and l not in old.splitlines())
    v = subprocess.run(["go", "vet", "./internal/dlq/"], cwd=w, env=env, capture_output=True, text=True)
    if v.returncode != 0:
        rows.append((name, "INVALID", changed, v.stderr.strip().splitlines()[-1][:80])); continue
    r = subprocess.run(["go", "test", "-count=1", "-timeout", "60s", "./internal/dlq/"], cwd=w, env=env, capture_output=True, text=True)
    out = r.stdout + r.stderr
    fails = sorted(set(re.findall(r"--- FAIL: (\S+)", out)))
    leaf = [f for f in fails if "/" in f] or fails
    rows.append((name, "KILLED" if r.returncode != 0 else "SURVIVED", changed, ",".join(leaf)[:110]))
k = sum(1 for r in rows if r[1] == "KILLED"); s = sum(1 for r in rows if r[1] == "SURVIVED"); i = len(rows) - k - s
print("SUMMARY: %d mutants: %d KILLED, %d SURVIVED, %d INVALID/BADPATTERN" % (len(rows), k, s, i))
for r in rows:
    if r[1] != "KILLED": print("  %-9s %s" % (r[1], r[0]))
print()
for r in rows:
    print("%-9s %s\n          mutant line: %s\n          failing: %s" % (r[1], r[0], r[2], r[3] or "-"))
