package dlq

// QA probe (not part of the PR): the page boundary that falls exactly at a partition's end, with a later non-empty partition.
import (
	"context"
	"reflect"
	"testing"
)

func TestQAProbePageEndsAtPartitionEnd(t *testing.T) {
	cases := []struct {
		name      string
		parts     map[int32][2]int64
		opt       ListOptions
		wantPages [][]string
		wantNext1 string
	}{
		{"P1 page fills at end of p0, p1 non-empty (Low 0)", map[int32][2]int64{0: {0, 3}, 1: {0, 3}}, ListOptions{PageSize: 3}, [][]string{span(0, 0, 3), span(1, 0, 3)}, "1/0"},
		{"P2 page fills at end of p0, p1 Low 5", map[int32][2]int64{0: {0, 3}, 1: {5, 2}}, ListOptions{PageSize: 3}, [][]string{span(0, 0, 3), span(1, 5, 7)}, "1/5"},
		{"P3 page fills at end of p0, p1 empty, p2 Low 4", map[int32][2]int64{0: {0, 2}, 1: {9, 0}, 2: {4, 3}}, ListOptions{PageSize: 2}, [][]string{span(0, 0, 2), span(2, 4, 6), span(2, 6, 7)}, "2/4"},
	}
	for _, tc := range cases {
		t.Run(tc.name, func(t *testing.T) {
			f := newFakeReader(tc.parts)
			m := NewManager(nil, f)
			opt := tc.opt
			var pages [][]string
			var warns []string
			for i := 0; i < 10; i++ {
				page, err := m.List(context.Background(), "t.dlq", opt)
				if err != nil {
					t.Fatalf("page %d: %v", i, err)
				}
				if i == 0 && page.Next.String() != tc.wantNext1 {
					t.Errorf("page 0 Next = %v (set %v), want %s", page.Next, page.Next.IsSet(), tc.wantNext1)
				}
				pages, warns = append(pages, po(page.Records)), append(warns, page.Warnings...)
				if !page.Next.IsSet() {
					break
				}
				opt.From, opt.Snapshot = page.Next, page.Watermarks
			}
			if !reflect.DeepEqual(pages, tc.wantPages) {
				t.Errorf("pages = %v, want %v", pages, tc.wantPages)
			}
			if len(warns) != 0 {
				t.Errorf("warnings = %q, want none (no offsets expired)", warns)
			}
		})
	}
}

// --from P/O exactly at Low (Low > 0) is not an expiry.
func TestQAProbeFromAtLowNoWarning(t *testing.T) {
	f := newFakeReader(map[int32][2]int64{0: {4, 3}})
	page, err := NewManager(nil, f).List(context.Background(), "t.dlq", ListOptions{From: Cursor{Partition: 0, Offset: 4, set: true}})
	if err != nil || len(page.Warnings) != 0 || !reflect.DeepEqual(po(page.Records), span(0, 4, 7)) {
		t.Errorf("records %v warnings %q err %v; want 0/4..0/6, no warning", po(page.Records), page.Warnings, err)
	}
}
