// Stub of github.com/oklog/ulid/v2 (signatures used by shared/events only).
package ulid

import (
	"fmt"
	"io"
	"time"
)

type ULID [16]byte

func (u ULID) String() string                     { return fmt.Sprintf("%x", u[:]) }
func Monotonic(r io.Reader, inc uint64) io.Reader { return r }
func Timestamp(t time.Time) uint64                { return uint64(t.UnixMilli()) }
func MustNew(ms uint64, r io.Reader) ULID {
	var u ULID
	for i := 0; i < 6; i++ {
		u[i] = byte(ms >> (8 * (5 - i)))
	}
	_, _ = io.ReadFull(r, u[6:])
	return u
}

===== mutate.sh =====
#!/bin/bash
# mutate.sh <app tree>: for each mutant, fresh harness module, apply, run full crawler package and -run '^Killer$'.
H=$(cd "$(dirname "$0")" && pwd); W=$1; M=$H/../mutmod
export GOTOOLCHAIN=local GOFLAGS=-mod=mod GOPROXY=off PATH=/usr/local/go1.25.1/bin:$PATH
n=$(python3 -c "exec(open('$H/mutants.py').read()); print(len(M))")
for i in $(seq 0 $((n-1))); do
  $H/build.sh $W $M
  info=$(python3 - "$H/mutants.py" "$M/internal/crawler" $i <<'PY'
import sys
exec(open(sys.argv[1]).read()); m = M[int(sys.argv[3])]; name, killer, old, new = m[:4]
p = sys.argv[2] + "/" + (m[4] if len(m) > 4 else "loop.go"); s = open(p).read()
assert s.count(old) == 1, "old text not unique/absent: " + name
open(p, "w").write(s.replace(old, new)); print(name + "|" + killer)
PY
) || { echo "APPLY FAILED $i"; continue; }
  name=${info%|*}; killer=${info#*|}
  (cd $M && go vet ./internal/crawler/ >/dev/null 2>&1) || { echo "$name: DOES NOT COMPILE"; continue; }
  full=$(cd $M && go test -count=1 ./internal/crawler/ 2>&1 | grep -- '^--- FAIL' | sed 's/--- FAIL: //; s/ (.*//' | tr '\n' ' ')
  one=$(cd $M && go test -count=1 -run "^${killer}\$" ./internal/crawler/ >/dev/null 2>&1 && echo SURVIVED || echo KILLED)
  echo "$name"
  echo "    full run failing tests: ${full:-NONE (SURVIVED)}"
  echo "    -run '^${killer}\$': $one"
done

[exit code 0]
