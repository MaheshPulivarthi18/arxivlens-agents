"""AC1/AC2 harness (QA scratch, not committed): runs the real test function from a given file on hand-made rows.
psycopg / pytest / services.common.config are stubbed in sys.modules; the cursor returns the rows below."""
import sys, types, importlib.util

ROWS = None
class Cur:
    def __enter__(self): return self
    def __exit__(self, *a): return False
    def execute(self, sql, *a): self.sql = sql
    def fetchall(self): return ROWS
class Conn:
    def cursor(self): return Cur()
    def close(self): pass
psy = types.ModuleType("psycopg"); psy.connect = lambda *a, **k: Conn()
sys.modules["psycopg"] = psy
sys.modules["pytest"] = types.ModuleType("pytest")
for name in ("services", "services.common"):
    sys.modules[name] = types.ModuleType(name)
cfg = types.ModuleType("services.common.config"); cfg.config = types.SimpleNamespace(postgres_url="x", embedding_model="m")
sys.modules["services.common.config"] = cfg

# (arxiv_id, version, chunk_index, page_number, page_start)
CASES = [
    ("None page_start",               [("2401.1", 1, 0, 1, None)]),
    ("page_number None (ps=3)",       [("2401.1", 1, 0, None, 3)]),
    ("page_start<page_number (1<2)",  [("2401.1", 1, 0, 2, 1)]),
    ("decreasing page_start",         [("2401.1", 1, 0, 1, 4), ("2401.1", 1, 1, 1, 2)]),
    ("NULL page_number + decreasing", [("2401.1", 1, 0, 1, 4), ("2401.1", 1, 1, None, 2)]),
    ("NULL pn then decreasing",       [("2401.1", 1, 0, None, 4), ("2401.1", 1, 1, 1, 2)]),
    ("all good",                      [("2401.1", 1, 0, 1, 1), ("2401.1", 1, 1, 1, 3), ("2401.2", 2, 0, 2, 2)]),
    ("empty result",                  []),
]

def run(path):
    global ROWS
    spec = importlib.util.spec_from_file_location("m2", path); m = importlib.util.module_from_spec(spec); spec.loader.exec_module(m)
    out = []
    for label, rows in CASES:
        ROWS = rows
        try:
            m.test_embedded_chunks_have_recorded_start_pages(); out.append((label, "PASS", ""))
        except AssertionError as e:
            out.append((label, "FAIL", str(e)))
        except Exception as e:
            out.append((label, "ERROR", f"{type(e).__name__}: {e}"))
    return out

if __name__ == "__main__":
    results = {p: run(p) for p in sys.argv[1:]}
    for p, res in results.items():
        print(f"== {p.rsplit('/',1)[-1]}")
