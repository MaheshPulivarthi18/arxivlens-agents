package query

// QA AI-8a probes (not part of the PR). Each line: PROBE <name>: <observed>.
import (
	"context"
	"encoding/json"
	"errors"
	"fmt"
	"testing"

	"arxivlens/internal/llm/llmtest"
)

func TestQAProbeAI8a(t *testing.T) {
	captureLog(t)
	ctx := context.Background()
	show := func(name string, resp *AskResponse) {
		out := []string{}
		for _, c := range resp.Citations {
			out = append(out, fmt.Sprintf("{m%d %s sup=%v score=%v quote=%q}", c.Marker, c.Status, c.Supported, c.Score, c.Quote))
		}
		fmt.Printf("PROBE %-58s kind=%q reason=%q %v\n", name+":", resp.AnswerKind, resp.RefusalReason, out)
	}
	llm := func(hits []SearchHit, v verifier, reply string) *AskResponse {
		r, _ := llmRAG(hits, v, llmtest.Reply(reply))
		resp, _ := r.Ask(ctx, "q", nil, 3)
		return resp
	}
	// P1 ADR-0014 case-sensitivity: the model quote differs from the chunk only in case.
	show("P1 model quote differs only in case (NLI 0.9)", llm(batchHits(), simVerifier(constSim(0.9)),
		`{"claims": [{"text": "It raises accuracy.", "citations": [1, 2], "quote": "THE PROPOSED METHOD improves accuracy"}]}`))
	// P2 Q4 normalisation on the chunk side: chunk has a newline, double space and an inline [12] inside the quoted span.
	h := batchHits()
	h[0].Content = "Paper — Method\n\nOur   model\nreaches 91% accuracy [12] on the held-out benchmark split."
	show("P2 quote vs chunk with newline/double space/[12]", llm(h, simVerifier(constSim(0.9)),
		`{"claims": [{"text": "It reaches 91%.", "citations": [1], "quote": "Our model reaches 91% accuracy on the held-out"}]}`))
	// P3 verifier down + fabricated model quote: dev note says unsupported (quote check ran and failed).
	show("P3 verifier error + fabricated quote", llm(batchHits(), &fakeVerifier{score: func(string, string) float64 { return 0.9 }, err: errors.New("down")},
		`{"claims": [{"text": "It raises accuracy.", "citations": [1], "quote": "accuracy doubles"}]}`))
	// P4 the same marker cited by two claims with different outcomes (ordering invariant, design).
	show("P4 marker 1 cited by a fabricated-quote claim and a good claim", llm(batchHits(), simVerifier(constSim(0.9)),
		`{"claims": [{"text": "It doubles.", "citations": [1, 2], "quote": "The Results section shows"}, {"text": "It raises accuracy.", "citations": [1]}]}`))
	// P5 extractive quote from a sentence with an inline reference: quote invariant must hold against raw content.
	h5 := []SearchHit{ragHit(1, "Method", 0.9)}
	h5[0].Content = "Paper — Method\n\nThe proposed method improves accuracy on every benchmark [3, 4] we tried in this study."
	r5 := newRAG(h5, nil)
	resp5, _ := r5.Ask(ctx, "q", nil, 3)
	show("P5 extractive, sentence with inline [3, 4]", resp5)
	if len(resp5.Citations) > 0 {
		c := resp5.Citations[0]
		fmt.Printf("PROBE %-58s inPassage(quote, content)=%v\n", "P5 invariant:", c.Quote != "" && inPassage(c.Quote, c.Content))
	}
	// P6 JSON of an unverified refusal-evidence citation: status present, page absent, quote absent.
	ev := refusal(ctx, "t", 0, citationCheck{hitScore: make([]float64, 1), hitScored: make([]bool, 1)}, []SearchHit{ragHit(1, "Method", 0.9)})
	raw, _ := json.Marshal(ev.Citations[0])
	fmt.Printf("PROBE %-58s %s\n", "P6 refusal evidence JSON:", raw)
}
