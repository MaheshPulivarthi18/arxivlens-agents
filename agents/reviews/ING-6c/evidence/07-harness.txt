def tree():
    d = tempfile.mkdtemp(prefix="m-", dir=os.path.join(S, "mut3"))
    subprocess.run(f"git -C {HEAD} archive HEAD | tar -x -C {d}", shell=True, check=True)
    return d
def run(d, suite):
    r = subprocess.run(SUITES[suite], cwd=d, env=ENV, capture_output=True, text=True, timeout=240)
    out = r.stdout + r.stderr
    if "[build failed]" in out or "[setup failed]" in out or "\n# " in "\n" + out and r.returncode != 0 and "--- FAIL" not in out and "FAIL\t" not in out.replace("[build failed]", ""):
        return "INVALID", out
    fails = sorted({l.strip().split(" ")[2] for l in out.splitlines() if l.strip().startswith("--- FAIL:")})
    if "panic: test timed out" in out: fails.append("TIMEOUT(hang: pool Close waits on a held conn)")
    return ("PASS" if r.returncode == 0 else "FAIL " + ",".join(fails)), out
