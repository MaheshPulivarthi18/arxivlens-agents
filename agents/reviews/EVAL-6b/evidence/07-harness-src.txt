import os, subprocess, sys, tempfile, difflib

WT = sys.argv[1]
SA = "scripts/evaluate_answers.py"
GUARD = '    if reason is not None and not isinstance(reason, str):\n        return "error"  # a non-string reason (even a falsy [] or 0) fails closed, never "no reason"\n'
K_CLASSIFY = "classify_non_string or classify_null_or_empty"
K_EVAL = "evaluate_completes_on_non_string"
K_MAIN = "test_main_"
K_AC4 = "failed_refusal_set_call"
MUTANTS = [
    ("m1 AC6: remove non-string guard in classify()", GUARD, "", K_CLASSIFY),
    ("m2 AC6: falsy non-string ([] / 0) falls through as no reason", "if reason is not None and not isinstance(reason, str):",
     "if reason and not isinstance(reason, str):", K_CLASSIFY),
    ("m3 AC6: count non-string values via str(reason)", "reason = _reason_str(resp or {})",
     'reason = str((resp or {}).get("refusal_reason") or "")', K_EVAL),
    ("m4 AC6: main always loads near-miss", "load_cases(args.near_miss) if args.near_miss else None",
     "load_cases(args.near_miss)", K_MAIN),
    ("m5 extra: case_record keeps raw non-string reason", '"refusal_reason": _reason_str(r) or None',

    ("m7 extra: guard drops `is not None` (null -> error)", "if reason is not None and not isinstance(reason, str):",

    ("m8 extra: _reason_str passes non-strings through", "return reason if isinstance(reason, str) else \"\"",
     'return reason or ""', K_EVAL),
    ("m9 extra: main swaps answers/refusals files", "evaluate(load_cases(args.answers), load_cases(args.refusals),",
     "evaluate(load_cases(args.refusals), load_cases(args.answers),", K_MAIN),
    ("m10 extra: main never writes --json-out", "    if args.json_out:\n", "    if False:\n", K_MAIN),

     K_CLASSIFY),

def run(d, *args):
    p = subprocess.run([sys.executable, "-m", "pytest", "-q", "-p", "no:cacheprovider", "tests/test_evals_data.py", *args],
                       cwd=d, capture_output=True, text=True)
    last = [l for l in p.stdout.splitlines() if l.strip()][-1:] or [""]
    return p.returncode, last[0]

def verdict(rc):
    return {0: "SURVIVED", 1: "KILLED"}.get(rc, f"INVALID(rc={rc})")

rows = []

    path = os.path.join(d, SA)
    src = open(path, encoding="utf-8").read()
    n = src.count(old)
    if n != 1:
        rows.append((name, f"INVALID(anchor x{n})", "", "", "")); continue
    mut = src.replace(old, new)
    open(path, "w", encoding="utf-8").write(mut)

        print(f"      {l.strip()[:120]}")
