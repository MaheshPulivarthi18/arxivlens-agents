Reads files via `git show <rev>:<path>` in the cwd (the PR worktree)."""
import subprocess, re, sys
from packaging.requirements import Requirement
BASE, HEAD = "40dec5a", "95ea3bb"
def show(rev, path):
    return subprocess.run(["git","show",f"{rev}:{path}"],capture_output=True,text=True,check=True).stdout
def flatten(rev, path, seen=None):
    seen = seen or []
    out = []
    for line in show(rev, path).splitlines():
        line = line.split("#",1)[0].strip()
        if not line: continue
        if line.startswith("-r "):
            out += flatten(rev, line[3:].strip(), seen + [path]); continue
        out.append(str(Requirement(line)))
    return out
def args(rev, tokens):
    s = []
    i = 0
    while i < len(tokens):
        t = tokens[i]
        if t == "-r": s += flatten(rev, tokens[i+1]); i += 2; continue
        if t.startswith("-"): i += 2 if t in ("--extra-index-url",) else 1; continue
        s.append(str(Requirement(t))); i += 1
    return s
def norm(lst):
