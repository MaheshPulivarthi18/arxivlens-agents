// QA probe rows (harness only, never pushed): the shape the missing locateQuote rows would take.
import { describe, it, expect } from 'vitest';
import { locateQuote, normQuote } from '__SRC__/lib/citations.js';
describe('QA gap rows', () => {
  it.each([
    // quote does not start at 0 and something before it normalises away: the raw start must be mapped, not reused
    ['ref + double space before the quote', 'As shown [12]  earlier, hybrid\nsearch works well.', 'hybrid search works', 'hybrid\nsearch works'],
    ['collapsed whitespace before the quote', 'Intro.\n\n\nWe  verify  each claim.', 'We verify each claim.', 'We  verify  each claim.'],
    // strings.Fields is Unicode: NBSP / U+2009 between words collapse to one space on the server
    ['NBSP between words', 'hybrid search works', 'hybrid search works', 'hybrid search works'],
    ['thin space U+2009', 'a b c', 'a b c', 'a b c'],
  ])('%s', (_n, content, quote, raw) => {
    const r = locateQuote(content, quote);
    expect(r === null ? null : content.slice(r.start, r.end)).toBe(raw);
    expect(normQuote(raw)).toBe(normQuote(quote));
  });
  it('whitespace-only quote on content that contains that whitespace → null', () => {
    expect(locateQuote('a \n b', ' \n ')).toBeNull();
  });
});

#!/usr/bin/env bash
# usage: mutate.sh <app frontend/src>  copies src to a fresh temp dir per mutant, applies it, runs the lib tests
set -u
SRC=$1; H=$(cd "$(dirname "$0")" && pwd)
N=$(python3 -I -c "import sys; sys.path.insert(0,'$H'); from mutants import MUTANTS; print(len(MUTANTS))")
for i in $(seq 0 $((N-1))); do
  T=$(mktemp -d); cp -r "$SRC" "$T/src"
  NAME=$(python3 -I - "$H" "$T/src/lib/citations.js" "$i" <<'PY'
import sys; sys.path.insert(0, sys.argv[1]); from mutants import MUTANTS
name, old, new = MUTANTS[int(sys.argv[3])]; p = sys.argv[2]; s = open(p).read()
assert s.count(old) == 1, f"{name}: pattern found {s.count(old)}x"
open(p, "w").write(s.replace(old, new)); print(name)
PY
) || { echo "APPLY FAILED $i"; continue; }
  OUT=$("$H/run.sh" "$T/src" mut 2>&1)
  if echo "$OUT" | grep -q "FAIL "; then
    echo "KILLED   $NAME"; echo "$OUT" | grep "FAIL " | head -2 | sed 's/^/           /'
  else echo "SURVIVED $NAME"; echo "$OUT" | sed 's/^/           /'; fi

// Minimal vitest stand-in for plain-ESM lib tests (logic check only; CI Vitest is the pass/fail).
import { isDeepStrictEqual, format } from 'node:util';
const results = []; const stack = [];
export const describe = (n, f) => { stack.push(n); f(); stack.pop(); };
const run = (name, f) => results.push({ name: [...stack, name].join(' > '), f });
export const it = Object.assign(run, {
  each: (rows) => (name, f) => rows.forEach((r) => { const a = Array.isArray(r) ? r : [r]; run(format(name, ...a), () => f(...a)); }),
});
export const test = it;
const fail = (m) => { throw new Error(m); };
const show = (v) => { try { return JSON.stringify(v); } catch { return String(v); } };
function matchers(v, neg) {
  const chk = (ok, m) => { if (ok === neg) fail((neg ? 'NOT ' : '') + m); };
  return {
    toBe: (e) => chk(Object.is(v, e), `toBe ${show(v)} vs ${show(e)}`),
    toEqual: (e) => chk(isDeepStrictEqual(JSON.parse(JSON.stringify(v ?? null)), JSON.parse(JSON.stringify(e ?? null))), `toEqual ${show(v)} vs ${show(e)}`),
    toBeNull: () => chk(v === null, `toBeNull ${show(v)}`),
    toMatch: (re) => chk(typeof re === 'string' ? String(v).includes(re) : re.test(v), `toMatch ${show(v)} ${re}`),
    toContain: (e) => chk(v != null && v.includes(e), `toContain ${show(v)} ${show(e)}`),
    toHaveLength: (n) => chk(v != null && v.length === n, `toHaveLength ${show(v)} ${n}`),
    toBeGreaterThan: (n) => chk(v > n, `toBeGreaterThan ${v} ${n}`),
  };
}
export const expect = (v) => Object.assign(matchers(v, false), { not: matchers(v, true) });
export const vi = {};
export async function report(label) {
  let pass = 0; const fails = [];
  for (const t of results) { try { await t.f(); pass += 1; } catch (e) { fails.push(`${t.name}: ${e.message.slice(0, 160)}`); } }
  console.log(`${label}: ${pass} passed, ${fails.length} failed`);
  for (const f of fails) console.log('  FAIL ' + f);
  return fails.length;
}
