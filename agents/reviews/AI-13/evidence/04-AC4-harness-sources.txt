def run(label, old, new):
    d = tempfile.mkdtemp(prefix="qa-ai13-mut-")
    subprocess.run(f"git -C {APP} archive HEAD | tar -x -C {d}", shell=True, check=True)
    p = os.path.join(d, "services/embedder/worker.py")
    src = open(p).read()
    n = src.count(old) if old else -1
    if old == "BASE":
        subprocess.run(f"git -C {APP} show origin/main:services/embedder/worker.py > {p}", shell=True, check=True)
    elif old is not None:
        if n != 1:
            return label, f"INVALID (anchor matched {n}x)", ""
        open(p, "w").write(src.replace(old, new))
    env = dict(os.environ, PYTHONPATH=f"{d}:{PLUGIN}")
    r = subprocess.run([sys.executable, "-m", "pytest", "-p", "qa_stubs", "-p", "no:cacheprovider", "-q", *TESTS],
                       cwd=d, env=env, capture_output=True, text=True)
    tail = r.stdout.strip().splitlines()[-1] if r.stdout.strip() else r.stderr[-200:]
    status = {0: "SURVIVED", 1: "KILLED"}.get(r.returncode, f"INVALID (pytest exit {r.returncode})")
    return label, status, tail


if __name__ == "__main__":
    ctrl = [run("C1 control: unmutated PR head (expect 2 passed)", None, None),
            run("C2 control: origin/main worker.py + PR tests (expect fail)", "BASE", None)]
    rows = ctrl + [run(*m) for m in MUTANTS]
    print(f"{'mutant':<64} {'result':<10} pytest summary")

    print(f"\n{k}/{len(rows) - 2} mutants killed. C1/C2 status column: SURVIVED = tests passed, KILLED = tests failed.")

"""QA AI-13 harness: turn every unimportable third-party module into a MagicMock module (logic checks only)."""
import importlib.abc, importlib.machinery, sys
from unittest.mock import MagicMock

STUB_ROOTS = {"torch", "psycopg", "psycopg_pool", "confluent_kafka", "fastapi", "ulid", "minio",
              "sentence_transformers", "transformers", "fitz", "pymupdf", "uvicorn", "tiktoken", "huggingface_hub",
              "grobid_client", "starlette", "prometheus_client", "boto3", "tokenizers", "pdfplumber", "httpx"}


class _Loader(importlib.abc.Loader):
    def create_module(self, spec):
        m = MagicMock(name=spec.name)
        m.__path__ = []
        m.__spec__ = spec
        m.__name__ = spec.name
        return m

    def exec_module(self, module):
        pass


class _Finder(importlib.abc.MetaPathFinder):
    def find_spec(self, name, path, target=None):
        if name.split(".")[0] not in STUB_ROOTS:
            return None
        if importlib.machinery.PathFinder.find_spec(name.split(".")[0]) is not None:
            return None  # real module available
        return importlib.machinery.ModuleSpec(name, _Loader(), is_package=True)


sys.meta_path.insert(0, _Finder())

import re, subprocess, sys, os
log, plugin = sys.argv[1], sys.argv[2]
lines = open(log).read().splitlines()
i = next(k for k, l in enumerate(lines) if "pytest -q tests/test_m0.py tests/test_verify.py tests/test_processor_pages.py" in l)
chars = ""
for l in lines[i + 1:]:
    m = re.match(r"^\S+Z ([.sFEx]+)\s+\[\s*\d+%\]$", l)

    chars += m.group(1)

"""Intersect ruff findings with lines added by the PR (git diff -U0 origin/main...HEAD). Run inside app/."""
import re, subprocess
files = ["services/embedder/worker.py", "tests/test_verify.py"]
added = {}
for f in files:
    d = subprocess.run(["git", "diff", "-U0", "origin/main...HEAD", "--", f], capture_output=True, text=True).stdout
    for m in re.finditer(r"^@@ -\S+ \+(\d+)(?:,(\d+))? @@", d, re.M):
        s, n = int(m.group(1)), int(m.group(2) or 1)
        added.setdefault(f, set()).update(range(s, s + n))
r = subprocess.run(["ruff", "check", "--isolated", "--select", "E,F,W", "--line-length", "130", "--output-format", "concise", *files],
                   capture_output=True, text=True)
finds = [l for l in r.stdout.splitlines() if re.match(r"^\S+:\d+:\d+:", l)]
on_added = [l for l in finds if int(l.split(":")[1]) in added.get(l.split(":")[0], set())]
print("added line ranges:", {f: f"{min(v)}-{max(v)} ({len(v)} lines)" for f, v in added.items()})
print(f"ruff findings total: {len(finds)}; on added lines: {len(on_added)}")
for l in on_added: print("  ON ADDED LINE:", l)
print("all findings (pre-existing lines):"); [print("  ", l) for l in finds]
