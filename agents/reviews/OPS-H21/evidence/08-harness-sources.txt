"""QA OPS-H21 probes (not committed to the PR): behaviours the ticket names but no committed test pins."""
from tests.test_arxiv_replay import Clock, FEED, PDF, rp  # noqa: F401  (reuses the PR's fixtures)
from tests.test_arxiv_replay import env  # noqa: F401
import urllib.request


def _run(tmp_path, *ids):
    argv = ["prefetch", "--dir", str(tmp_path / "s"), "--budget-seconds", "1000"]
    for i in ids:
        argv += ["--id", i]
    return rp.main(argv)


def test_probe_5xx_gets_another_pass(env, tmp_path, monkeypatch, capsys):
    up, _ = env
    up.script = {"api": [(200, FEED, {})], "pdf": [(503, b"", {})] * 4 + [(200, PDF, {})]}
    Clock(monkeypatch, up)
    assert _run(tmp_path, "2401.00001") == 0
    assert "retrying 1 queries" in capsys.readouterr().err


def test_probe_network_error_gets_another_pass(env, tmp_path, monkeypatch, capsys):
    up, _ = env
    up.script = {"api": [(200, FEED, {})], "pdf": [(200, PDF, {})]}
    Clock(monkeypatch, up)
    real, n = urllib.request.urlopen, []

    def flaky(req, timeout=None):
        if req.full_url.endswith(".pdf") and len(n) < 4:
            n.append(1)
            raise TimeoutError("simulated")
        return real(req, timeout=timeout)
    monkeypatch.setattr(rp.urllib.request, "urlopen", flaky)
    assert _run(tmp_path, "2401.00001") == 0
    assert "retrying 1 queries" in capsys.readouterr().err


def test_probe_query_with_404_and_429_pdf_is_retried(env, tmp_path, monkeypatch, capsys):
    feed2 = (b'<?xml version="1.0"?><feed xmlns="http://www.w3.org/2005/Atom">'
             b'<entry><id>p1</id></entry><entry><id>p2</id></entry></feed>')
    up, _ = env
    up.script = {"api": [(200, feed2, {})], "/pdf/p1.pdf": [(404, b"", {})],
                 "/pdf/p2.pdf": [(429, b"", {})] * 4 + [(200, PDF, {})], "pdf": [(404, b"", {})]}
    Clock(monkeypatch, up)
    assert _run(tmp_path, "2401.00001") == 1
    paths = [p for p, _ in up.requests]
    err = capsys.readouterr().err
    print("requests:", paths)
    assert paths.count("/pdf/p2.pdf") == 5, paths  # retried in pass 2 and saved
    assert "prefetch FAILED for 1: p1.pdf (arXiv" in err
    # the permanent p1.pdf is asked again in pass 2 because its query is re-run (query-level retry, per AC2)
    print("p1.pdf requests:", paths.count("/pdf/p1.pdf"))
