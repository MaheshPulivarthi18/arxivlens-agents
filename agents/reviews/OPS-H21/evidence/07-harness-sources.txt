def run(d, sel):
    args = [sys.executable, "-m", "pytest", "-q", "-p", "no:cacheprovider", T] + (["-k", sel] if sel != "-" else [])
    p = subprocess.run(args, cwd=d, capture_output=True, text=True, timeout=600)
    last = [l for l in p.stdout.splitlines() if l.strip()][-1]
    return p.returncode, last


rows = []
for mid, desc, target, subs in MUTANTS:
    d = tempfile.mkdtemp(prefix=mid + "-", dir=WORK)
    subprocess.run(f"git -C {APP} archive HEAD | tar -x -C {d}", shell=True, check=True)
    path = os.path.join(d, "scripts/arxiv_replay.py")
    src = open(path).read()
    ok = all(src.count(old) == 1 for old, _ in subs)
    if not ok:
        rows.append((mid, "INVALID (anchor not found exactly once)", "", desc)); continue
    for old, new in subs:
        src = src.replace(old, new)
    open(path, "w").write(src)
    diff = subprocess.run(["git", "diff", "--no-index", "--unified=0", os.path.join(APP, "scripts/arxiv_replay.py"), path],
                          capture_output=True, text=True).stdout
    changed = [l for l in diff.splitlines() if l[:1] in "+-" and not l.startswith(("+++", "---"))]
    tgt = run(d, target) if target != "-" else None
    full = run(d, "-")

                 f" | full file: {full[1]}", desc + "\n      diff: " + " / ".join(changed)))

print(f"{'id':4} {'verdict':9} result")
for mid, verdict, res, desc in rows:
    print(f"{mid:4} {verdict:9} {res}\n      {desc}")

"""QA OPS-H21 probes (not part of the PR): the halves of the transient rule that no PR test pins.
Copied into tests/ of a git-archive copy; reuses the PR's fixtures."""
from tests.test_arxiv_replay import EMPTY_FEED, PDF, Clock, env, rp  # noqa: F401  (env is a fixture)

A = "/api/query?id_list=2401.00001"
FEED2PDF = (b'<?xml version="1.0"?><feed xmlns="http://www.w3.org/2005/Atom">'
            b'<entry><id>p1</id></entry><entry><id>p2</id></entry></feed>')


def test_probe_5xx_gets_another_pass(env, tmp_path, monkeypatch):
    up, _ = env
    up.script = {A: [(503, b"", {})] * 4 + [(200, EMPTY_FEED, {})], "pdf": [(200, PDF, {})]}
    Clock(monkeypatch, up)
    assert rp.main(["prefetch", "--dir", str(tmp_path / "s"), "--id", "2401.00001", "--budget-seconds", "1000"]) == 0
    assert [p for p, _ in up.requests] == [A] * 5  # pass 1: 4 attempts; pass 2 succeeds


def test_probe_network_error_gets_another_pass(env, tmp_path, monkeypatch):
    up, _ = env
    up.script = {A: [(200, EMPTY_FEED, {})], "pdf": [(200, PDF, {})]}
    Clock(monkeypatch, up)
    real, calls = rp.urllib.request.urlopen, []

    def urlopen(req, timeout=None):  # the first 4 attempts time out (a real exception, mapped to 504 by fetch)
        calls.append(req.full_url)
        if len(calls) <= 4:
            raise TimeoutError("timed out")
        return real(req, timeout=timeout)
    monkeypatch.setattr(rp.urllib.request, "urlopen", urlopen)
    assert rp.main(["prefetch", "--dir", str(tmp_path / "s"), "--id", "2401.00001", "--budget-seconds", "1000"]) == 0
    assert len(calls) == 5


def test_probe_query_with_404_and_429_pdf_is_retried(env, tmp_path, monkeypatch):
    up, _ = env
    up.script = {A: [(200, FEED2PDF, {})], "/pdf/p1.pdf": [(404, b"", {})],
                 "/pdf/p2.pdf": [(429, b"", {})] * 4 + [(200, PDF, {})], "pdf": [(200, PDF, {})]}
    Clock(monkeypatch, up)
    assert rp.main(["prefetch", "--dir", str(tmp_path / "s"), "--id", "2401.00001", "--budget-seconds", "1000"]) == 1
    paths = [p for p, _ in up.requests]
    assert paths[:6] == [A, "/pdf/p1.pdf"] + ["/pdf/p2.pdf"] * 4  # pass 1
    assert "/pdf/p2.pdf" in paths[6:]  # pass 2 re-runs the query: the 429 PDF is asked again and saved
    assert rp.Recorder(str(tmp_path / "s"), False).load("pdf", "p2.pdf") == PDF
