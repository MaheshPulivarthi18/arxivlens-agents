func splitIntoSentences(text string) []string {
	var sentences []string
	var current strings.Builder
	for i := 0; i < len(text); i++ {
		ch := text[i]
		current.WriteByte(ch)
		if (ch == '.' || ch == '!' || ch == '?') && (i+1 == len(text) || text[i+1] == ' ') && !endsWithAbbreviation(current.String()) {
			s := strings.TrimSpace(current.String())
			if s != "" {
				sentences = append(sentences, s)
			}
			current.Reset()
			if i+1 < len(text) && text[i+1] == ' ' {
				i++ // skip space
			}
		}
	}
	if current.Len() > 0 {
		s := strings.TrimSpace(current.String())
		if s != "" {
			sentences = append(sentences, s)
		}
	}
	if len(sentences) == 0 && strings.TrimSpace(text) != "" {
		sentences = append(sentences, strings.TrimSpace(text))
	}
	return sentences
}

var sentenceAbbreviations = []string{"e.g.", "i.e.", "et al.", "cf.", "vs.", "fig.", "eq."}

func endsWithAbbreviation(s string) bool {
	lower := strings.ToLower(s)
	for _, abbr := range sentenceAbbreviations {
		if strings.HasSuffix(lower, abbr) {
			rest := lower[:len(lower)-len(abbr)]
			if rest == "" || strings.HasSuffix(rest, " ") || strings.HasSuffix(rest, "(") {
				return true
			}
		}
	}
	return false

SENTENCE_ABBREVIATIONS = ("e.g.", "i.e.", "et al.", "cf.", "vs.", "fig.", "eq.")
MARKER = re.compile(r"\[(\d+)\]")


def _ends_with_abbreviation(s: str) -> bool:
    lower = s.lower()
    for abbr in SENTENCE_ABBREVIATIONS:
        if lower.endswith(abbr):
            rest = lower[:-len(abbr)]
            if rest == "" or rest.endswith(" ") or rest.endswith("("):
                return True
    return False


def split_sentences(text: str) -> List[str]:
    """Split like rag.go: end at '.', '!' or '?' followed by a space or the end, unless an abbreviation."""
    sentences: List[str] = []
    cur = ""
    i = 0
    while i < len(text):
        cur += text[i]
        if text[i] in ".!?" and (i + 1 == len(text) or text[i + 1] == " ") and not _ends_with_abbreviation(cur):
            if cur.strip():
                sentences.append(cur.strip())
            cur = ""
            if i + 1 < len(text):
                i += 1  # skip the space
        i += 1
    if cur.strip():
        sentences.append(cur.strip())
    return sentences
