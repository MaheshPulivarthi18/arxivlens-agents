#!/usr/bin/env python3
"""QA AI-8c mutation probe. For each single-edit mutant of internal/query/rag.go (LLM synthesis path), run the
8 Scope tests with synthesis_test.go at (a) origin/main (no invariant calls) and (b) the PR head (11 calls).
A mutant that does not compile is INVALID, never KILLED. Usage: mutate.py <harness-module-dir> <base synthesis_test.go>"""
import os, shutil, subprocess, sys
mod, base_test = sys.argv[1], sys.argv[2]
rag = os.path.join(mod, "internal/query/rag.go")
st = os.path.join(mod, "internal/query/synthesis_test.go")
orig_rag, head_test = open(rag).read(), open(st).read()
base = open(base_test).read()
RUN = "^(TestSynthesisAnswersWithVerifiedClaims|TestSynthesisDropsUnsupportedClaims|TestSynthesisRefusals|TestSynthesisFallsBackToExtractive|TestSynthesisClaimsUnsupportedFallBackToQuotes|TestQuickAnswerSkipsSynthesis|TestSynthesisMarkersFollowPromptOrder|TestWithLLMNilKeepsExtractive)$"
MUTANTS = [
  ("M0 control (no edit)", None, None),
  ("M1 quote on unsupported citation (drop `case !cite.Supported:`)", "\t\tcase !cite.Supported:\n", "\t\tcase false:\n"),
  ("M2 supported != status (force Supported=true after newCitation)", "\t\tcite := newCitation(rf.marker, passages[h].Hit, score, status)\n", "\t\tcite := newCitation(rf.marker, passages[h].Hit, score, status)\n\t\tcite.Supported = true\n"),
  ("M3 invalid status on LLM path (status=\"ok\")", "\t\tstatus := citationStatus(score, ok)\n", "\t\tstatus := citationStatus(score, ok)\n\t\tif status == StatusSupported {\n\t\t\tstatus = \"ok\"\n\t\t}\n"),
  ("M4 non-verbatim quote (exact-match quote := claim text + \" (paraphrased)\")", "\t\t\tcite.Quote = normQuote(c.Text)\n", "\t\t\tcite.Quote = normQuote(c.Text) + \" (paraphrased)\"\n"),
  ("M5 refusal evidence marked supported (refusal(): Supported=true)", "\t\tevidence[i] = newCitation(i+1, h, check.hitScore[i], status) // 0 when no checked citation scored hit i\n", "\t\tevidence[i] = newCitation(i+1, h, check.hitScore[i], status) // 0 when no checked citation scored hit i\n\t\tevidence[i].Supported = true\n"),
  ("M6 quoted path: quote on unsupported citation (`exact[pi] && cite.Supported` -> `exact[pi]`)", "\t\tif exact[pi] && cite.Supported {\n", "\t\tif exact[pi] {\n"),
  ("M7 quoted path: non-verbatim quote (sentence + \" (sic)\")", "\t\t\tcite.Quote = normQuote(sentTexts[c.sentIdx])\n", "\t\t\tcite.Quote = normQuote(sentTexts[c.sentIdx]) + \" (sic)\"\n"),
  ("M8 LLM path: invalid status on unsupported citations (\"rejected\")", "\t\tstatus := citationStatus(score, ok)\n", "\t\tstatus := citationStatus(score, ok)\n\t\tif status == StatusUnsupported {\n\t\t\tstatus = \"rejected\"\n\t\t}\n"),
  ("M9 quoted path: Supported=true on every citation", "\t\tvalidatedCitations = append(validatedCitations, cite)\n", "\t\tcite.Supported = true\n\t\tvalidatedCitations = append(validatedCitations, cite)\n"),
]
def run():
    p = subprocess.run(["go","test","-count=1","-run",RUN,"./internal/query/"], cwd=mod, capture_output=True, text=True,
        env={**os.environ,"GOTOOLCHAIN":"local","GOFLAGS":"-mod=mod","GOPROXY":"off"})
    out = p.stdout + p.stderr
    if "[build failed]" in out or "[setup failed]" in out: return "INVALID", out
    fails = sorted({l.split()[2] for l in out.splitlines() if l.strip().startswith("--- FAIL:") and l.startswith("--- FAIL")})
    return ("KILLED " + ",".join(fails)) if p.returncode else "SURVIVED", out
rows, logs = [], []
try:
    for name, old, new in MUTANTS:
        if old is not None:
            assert orig_rag.count(old) == 1, ("anchor not unique/absent", name)
            open(rag,"w").write(orig_rag.replace(old,new))
        res = {}
        for label, body in (("base tests (0 calls)", base), ("PR tests (11 calls)", head_test)):
            open(st,"w").write(body); res[label], out = run()
            logs.append(f"### {name} | {label}\n" + "\n".join(l for l in out.splitlines() if "breaks the AC1" in l or l.startswith(("--- FAIL","ok","FAIL")))[:1500])
        rows.append((name, res))
        open(rag,"w").write(orig_rag)
finally:
    open(rag,"w").write(orig_rag); open(st,"w").write(head_test)
print("RESULTS (go test -run <8 Scope tests> ./internal/query/; harness = PR-head internal/ + pgx/minio stubs, go1.24)")
for name, res in rows:
    print(f"- {name}")
    for k, v in res.items(): print(f"    {k:22s} {v}")
print("\nDETAIL (invariant messages / FAIL lines)")
print("\n".join(logs))
