#!/usr/bin/env bash
# ruff E,F,W findings on lines ADDED by the PR (ticket: "no findings on added lines"), plus totals main vs head.
LL=${LL:-130}; echo "== ruff E,F,W at line-length $LL"
for f in scripts/evaluate_answers.py tests/test_evals_data.py; do
  added=$(git diff -U0 origin/main...HEAD -- "$f" | awk '/^@@/{split($3,a,/[+,]/); s=a[2]; n=(a[3]==""?1:a[3]); for(i=s;i<s+n;i++) print i}')
  head_n=$(ruff check --select E,F,W --line-length $LL --output-format concise "$f" 2>/dev/null | grep -c "^$f:")
  main_n=$(git show origin/main:"$f" | ruff check --select E,F,W --line-length $LL --output-format concise --stdin-filename "$f" - 2>/dev/null | grep -c "^$f:")
  echo "$f: findings main=$main_n head=$head_n; added lines: $(echo $added | wc -w)"
  ruff check --select E,F,W --line-length $LL --output-format concise "$f" 2>/dev/null | grep "^$f:" | while IFS=: read -r p l c rest; do
    echo "$added" | grep -qx "$l" && echo "  ADDED-LINE FINDING $p:$l:$c:$rest"; done
  echo "  line-length $LL (no ruff config in repo); findings by code at head:"; ruff check --select E,F,W --line-length $LL --statistics "$f" 2>/dev/null | sed 's/^/    /'
done
