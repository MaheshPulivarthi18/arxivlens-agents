#!/usr/bin/env bash
# For each mutant: fresh module copy of the PR head (mkmod.sh), one anchored perl substitution on server.go,
# check the diff is non-empty and touches code (printed in the diff file), go vet (failure => INVALID), then run
# (A) the AC3 command and (B) the whole internal/query package. KILLED only if the test run failed after a clean build.
H=/tmp/claude-0/-home-claude/c034de38-2b32-59cf-b61a-d5a273c9087b/scratchpad/qa-RET-1e-r1
export PATH=/usr/local/go1.25.1/bin:$PATH GOTOOLCHAIN=local GOFLAGS=-mod=mod GOPROXY=off GOSUMDB=off
SRC=${SRC:-/root/wt/qa-app-RET-1e}
OUT=${OUT:-$H/mutants-out}; mkdir -p $OUT; : > $OUT/diffs.txt
printf '%-4s %-12s %-12s %s\n' id "AC3-run" "pkg-full" mutant
while IFS='|' read -r id desc expr; do
  d=$(mktemp -d "$H/mut-$id-XXXX"); $H/bin/mkmod.sh "$SRC" "$d"
  f=$d/internal/query/server.go
  perl -0pi -e "$expr" "$f"
  if diff -q "$SRC/internal/query/server.go" "$f" >/dev/null; then printf '%-4s %-12s %-12s %s\n' "$id" NOT-APPLIED - "$desc"; continue; fi
  { echo "### $id: $desc"; diff "$SRC/internal/query/server.go" "$f"; echo; } >> $OUT/diffs.txt
  if ! (cd "$d" && go vet ./internal/query/ >"$d/build.log" 2>&1); then printf '%-4s %-12s %-12s %s\n' "$id" INVALID INVALID "$desc"; continue; fi
  (cd "$d" && go test -count=1 -run '^(TestWriteJSON|TestHandlersEncodeContract|TestHandlersErrorContract)$' ./internal/query/ >"$d/testA.log" 2>&1); a=$?
  (cd "$d" && go test -count=1 ./internal/query/ >"$d/testB.log" 2>&1); b=$?
  va=$([ $a -ne 0 ] && echo KILLED || echo SURVIVED); vb=$([ $b -ne 0 ] && echo KILLED || echo SURVIVED)
  rows=$(grep -E -- '--- FAIL: TestHandlersErrorContract/' "$d/testA.log" | sed 's/ (.*//;s/.*TestHandlersErrorContract\///' | sort -u | paste -sd, -)
  printf '%-4s %-12s %-12s %s%s\n' "$id" "$va" "$vb" "$desc" "${rows:+  [ErrorContract rows failing: $rows]}"
done < ${MUTANTS:-$H/bin/mutants.txt}
