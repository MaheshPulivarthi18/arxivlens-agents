    s, l = s / 100, l / 100

        return l - a * max(-1, min(k - 3, 9 - k, 1))
    return (f(0), f(8), f(4))
def lin(v): return v / 12.92 if v <= 0.04045 else ((v + 0.055) / 1.055) ** 2.4

// QA FE-0e r1: minimal vitest API stub for running tokens.test.js under plain Node (no Vite transform!).
// It answers "does the guard LOGIC detect palette classes"; it can NOT reproduce Vite's
// new URL(`...${x}`, import.meta.url) rewrite that breaks the test in real vitest (see CI log).
import { isDeepStrictEqual, inspect } from 'node:util'
export const results = []
const run = (name, fn, args) => { try { fn(...args); results.push(['PASS', name]) } catch (e) { results.push(['FAIL', name, e.message]) } }
export const describe = (_n, fn) => fn()
export const it = (name, fn) => run(name, fn, [])
it.each = (rows) => (name, fn) => { for (const r of rows) run(name.replace('%s', r), fn, Array.isArray(r) ? r : [r]) }
export const expect = (v) => {
  const m = (neg) => ({
    toMatch: (re) => { if (re.test(v) === neg) throw new Error(`${inspect(v)} ${neg ? 'matched' : 'did not match'} ${re}`) },
    toBeGreaterThan: (n) => { if ((v > n) === neg) throw new Error(`${v} > ${n} is ${!neg}`) },
    toEqual: (w) => { if (isDeepStrictEqual(v, w) === neg) throw new Error(`expected ${inspect(w)}, got ${inspect(v)}`) },
  })
  return { ...m(false), not: m(true) }
}
