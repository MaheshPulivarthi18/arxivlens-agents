#!/bin/bash
# QA OPS-H5b r1: run the PR's real search_integration_test.go (+ real internal/db, real pgx v5.6.0) against local PG16.
# usage: run.sh <module dir>   (DB: migrations 000001+000003 applied, no pgvector)
M=$1; R=$(dirname "$0"); PGD=$(cat $R/pgdir)
export GOTOOLCHAIN=local GOPROXY=off GOSUMDB=off GOFLAGS=-mod=mod
export PGOPTIONS="-c client_min_messages=warning"
export DATABASE_URL="postgres://postgres@/arxivlens?host=$PGD&sslmode=disable"
q(){ psql -h $PGD -U postgres -d arxivlens -qtAX -c "$1"; }
state(){ echo "   DB after: papers=[$(q "SELECT string_agg(arxiv_id, ',' ORDER BY arxiv_id) FROM papers")] chunks=$(q 'SELECT count(*) FROM paper_chunks')"; }
reset(){ q "TRUNCATE papers CASCADE"; }
t(){ # $1=CI value or '' ; $2=-run ; $3=count
  (cd $M && if [ -n "$1" ]; then export CI=$1; else unset CI; fi; go test -tags integration -count=${3:-1} -v -run "$2" ./internal/query/ 2>&1) \
   | grep -E -e '--- (PASS|FAIL|SKIP)' -e '_test.go:[0-9]+:' -e 'panic' | sed 's/^ */   /' | cut -c1-170; }
echo "S1 empty DB, CI unset: TestHNSWSeedCleanup -count=2"; reset; t "" '^TestHNSWSeedCleanup$' 2; state
echo "S2 dev DB (real paper + chunk, leftover synth.00001), CI unset: TestHNSWVectorHits"
reset; q "INSERT INTO papers(arxiv_id,title) VALUES ('2401.00001','Real'),('synth.00001','Leftover')"; q "INSERT INTO paper_chunks(paper_id,chunk_index,content,content_hash) SELECT id,0,'real text','h' FROM papers WHERE arxiv_id IN ('2401.00001','synth.00001')"
t "" '^TestHNSWVectorHits$'; state
echo "S3 same dev DB, CI=true: TestHNSWVectorHits"; t true '^TestHNSWVectorHits$'; state
echo "S4 same dev DB, CI unset: TestHNSWSeedCleanup -count=2"; t "" '^TestHNSWSeedCleanup$' 2; state
echo "S5 empty DB, CI=true: TestHNSWVectorHits passes the guard (then fails: no pgvector here)"; reset; t true '^TestHNSWVectorHits$'; state

[exit code 0]
