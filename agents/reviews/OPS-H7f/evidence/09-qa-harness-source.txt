func TestQAReplayDryRunEqualsReal(t *testing.T) {
	for _, filter := range []string{"", "valueerror", "JSONDecodeError"} {
		dn, dpub, dlog := qaRun(t, filter, 100, true)
		rn, rpub, rlog := qaRun(t, filter, 100, false)
		t.Logf("filter=%q dry-run count=%d published=%d | real count=%d published=%d", filter, dn, len(dpub), rn, len(rpub))
		if len(dpub) != 0 {
			t.Errorf("dry-run published %d envelopes", len(dpub))
		}
		if dn != rn || rn != len(rpub) {
			t.Errorf("filter=%q: dry-run count %d, real count %d, real publishes %d: want all equal", filter, dn, rn, len(rpub))
		}

			}

// QA stub of github.com/oklog/ulid/v2 (only the API used by shared/events and the tests).
// Encodes 48-bit ms + 80-bit entropy as 26 Crockford base32 chars; Parse validates the same.
package ulid

import (
	"errors"
	"io"
	"math/big"
	"strings"
	"time"
)

const alphabet = "0123456789ABCDEFGHJKMNPQRSTVWXYZ"

type ULID [16]byte

type MonotonicEntropy struct{ r io.Reader }

func (m *MonotonicEntropy) Read(p []byte) (int, error) { return m.r.Read(p) }

func Monotonic(r io.Reader, inc uint64) *MonotonicEntropy { return &MonotonicEntropy{r: r} }

func Timestamp(t time.Time) uint64 { return uint64(t.UnixMilli()) }

func MustNew(ms uint64, entropy io.Reader) ULID {
	var u ULID
	for i := 0; i < 6; i++ {
		u[i] = byte(ms >> (8 * (5 - i)))
	}
	if _, err := io.ReadFull(entropy, u[6:]); err != nil {
		panic(err)
	}
	return u
}

func (u ULID) String() string {
	n := new(big.Int).SetBytes(u[:])
	out := make([]byte, 26)
	b32 := big.NewInt(32)
	m := new(big.Int)
	for i := 25; i >= 0; i-- {
		n.DivMod(n, b32, m)
		out[i] = alphabet[m.Int64()]
	}
	return string(out)
}

var ErrDataSize = errors.New("ulid: bad data size when unmarshaling")
var ErrInvalidCharacters = errors.New("ulid: bad data characters when unmarshaling")
var ErrOverflow = errors.New("ulid: overflow when unmarshaling")

func Parse(s string) (ULID, error) {
	var u ULID
	if len(s) != 26 {
		return u, ErrDataSize
	}
	for _, c := range s {
		if !strings.ContainsRune(alphabet, c) {
			return u, ErrInvalidCharacters
		}
	}
	if s[0] > '7' {
		return u, ErrOverflow
	}
	return u, nil
}
