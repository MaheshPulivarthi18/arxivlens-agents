# Single-edit mutants of the step's script; each is run on every store and compared to the original's output.
import subprocess, sys, os, glob
H = os.path.dirname(os.path.abspath(__file__)); src = open(f"{H}/risk1.py").read()
M = {
 "M1 compare a<b -> a>b": ("if a[1] < b[1]]", "if a[1] > b[1]]"),
 "M2 compare a<b -> a<=b (ties = violation)": ("if a[1] < b[1]]", "if a[1] <= b[1]]"),
 "M3 drop undated filter": ("dated += [(n, p)] if p else []", "dated += [(n, p)]"),
 "M4 n/a threshold <2 -> <1": ("if len(dated) < 2:", "if len(dated) < 1:"),
 "M5 sortBy filter inverted": ('if "sortBy=submittedDate" not in q:', 'if "sortBy=submittedDate" in q:'),
 "M6 report last violation": ("bad[0]", "bad[-1]"),
 "M7 drop .strip() on published": ('.findtext(A + "published") or "").strip()', '.findtext(A + "published") or "")'),
 "M8 no namespace": ('A = "{http://www.w3.org/2005/Atom}"', 'A = ""'),
}
def run(code, d):
    r = subprocess.run([sys.executable, "-I", "-c", code], cwd=d, capture_output=True, text=True)
    return r.stdout + r.stderr + f"exit={r.returncode}"
stores = sorted(glob.glob(f"{H}/stores/*/"))
base = {d: run(src, d) for d in stores}
for name, (a, b) in M.items():
    assert src.count(a) == 1, name
    mut = src.replace(a, b)
    diff = [os.path.basename(d.rstrip("/")) for d in stores if run(mut, d) != base[d]]
    ac2 = [s for s in diff if s[:2] in ("01", "02", "03", "04")]
    print(f"{name:45s} {'KILLED' if diff else 'SURVIVED'}  by AC2 stores: {ac2 or '-'}  all: {diff or '-'}")

#!/bin/bash
# (a) python3 that crashes / is killed -> the step's `|| echo` must keep the step at exit 0 under bash -eo pipefail (GitHub's default shell).
# (b) read-only: sha256 of every store file before == after; (c) no network: run inside an unshare'd netns if available.
H=$(cd "$(dirname "$0")" && pwd)
F=$(mktemp -d); mkdir -p "$F/bin"
printf '#!/bin/sh\necho "fake python3: simulated crash" >&2\nexit 1\n' > "$F/bin/python3"; chmod +x "$F/bin/python3"
(cd "$H/stores/01-asc" && PATH="$F/bin:$PATH" bash --noprofile --norc -eo pipefail "$H/step_run.sh"); echo "(a1) python3 exits 1 -> step exit=$?"
printf '#!/bin/sh\nkill -9 $$\n' > "$F/bin/python3"
(cd "$H/stores/01-asc" && PATH="$F/bin:$PATH" bash --noprofile --norc -eo pipefail "$H/step_run.sh"); echo "(a2) python3 SIGKILL -> step exit=$?"
before=$(cd "$H/stores" && find . -type f | sort | xargs sha256sum | sha256sum)
for d in "$H"/stores/*/; do (cd "$d" && bash --noprofile --norc -eo pipefail "$H/step_run.sh" >/dev/null 2>&1); done
after=$(cd "$H/stores" && find . -type f | sort | xargs sha256sum | sha256sum)
[ "$before" = "$after" ] && echo "(b) stores unchanged after 12 runs: $before" || echo "(b) STORES CHANGED"

if unshare -rn true 2>/dev/null; then (cd "$H/stores/02-desc" && unshare -rn python3 -I "$H/risk1.py" | tail -1); echo "(c) in empty netns exit=$?"; else echo "(c) unshare unavailable"; fi
=================== ./risk1.py
import glob, json, xml.etree.ElementTree as ET
A = "{http://www.w3.org/2005/Atom}"
found = 0
for meta in sorted(glob.glob(".arxiv-replay/api/*.json")):
    try:
        q = json.load(open(meta)).get("query", "")
        if "sortBy=submittedDate" not in q:
            continue
        found += 1
        print(f"RISK1 query={q}")
        dated = []
        for n, e in enumerate(ET.parse(meta[:-5]).getroot().iter(A + "entry"), 1):
            i, p = (e.findtext(A + "id") or "").split("/abs/", 1)[-1], (e.findtext(A + "published") or "").strip()
            print(f"RISK1 entry {n} {i} published={p}")
            dated += [(n, p)] if p else []
        bad = [(a, b) for a, b in zip(dated, dated[1:]) if a[1] < b[1]]
        if len(dated) < 2:
            print(f"RISK1 ORDER n/a ({len(dated)} entries)")
        elif bad:
            (i, p), (j, r) = bad[0]
            print(f"RISK1 ORDER NOT descending (first violation: entry {i} {p} < entry {j} {r})")
        else:
            print("RISK1 ORDER descending")
    except Exception as err:
        print(f"::warning::RISK1 {meta}: {type(err).__name__}: {err}")
if not found:
    print("RISK1 none: no recorded sortBy=submittedDate feed in .arxiv-replay/api")
    print("::warning::RISK1 none: no recorded sortBy=submittedDate feed in .arxiv-replay/api")
