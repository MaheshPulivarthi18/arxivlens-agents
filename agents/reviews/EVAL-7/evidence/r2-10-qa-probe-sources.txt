def run(name, path, old, new):
    d = subprocess.run(["mktemp", "-d", f"{OUT}/mut.XXXX"], capture_output=True, text=True).stdout.strip()
    for sub in ("scripts", "tests", "evals"):
        shutil.copytree(f"{APP}/{sub}", f"{d}/{sub}")
    src = open(f"{d}/{path}").read()
    if src.count(old) != 1:
        return "INVALID (text not found once)"
    open(f"{d}/{path}", "w").write(src.replace(old, new))
    p = subprocess.run([sys.executable, "-m", "pytest", "-q", "-p", "no:cacheprovider", "tests/test_evals_data.py",
                        "-k", "review"], cwd=d, capture_output=True, text=True)
    last = (p.stdout.strip().splitlines() or ["?"])[-1]
    return {0: "SURVIVED", 1: "KILLED"}.get(p.returncode, f"INVALID (pytest exit {p.returncode})") + "  | " + last

s = [n for n, r in results if r.startswith("SURVIVED")]
print(f"SUMMARY: {k}/{len(results)} killed; survived: {s or 'none'}\n")
for i, (n, r) in enumerate(results, 1):
