def run(d, suite):
    subprocess.run(["psql", "-h", "/tmp/qapg", "-p", "55432", "-U", "postgres", "-d", "arxivlens", "-qc",
                    "DELETE FROM dlq_replay_log; DELETE FROM dlq_replay_checkpoints"], capture_output=True)
    r = subprocess.run(SUITES[suite], cwd=d, env=ENV, capture_output=True, text=True, timeout=300)
    out = r.stdout + r.stderr
    if "[build failed]" in out or "[setup failed]" in out:
        return "INVALID", out
    fails = sorted({l.strip().split(" ")[2] for l in out.splitlines() if l.strip().startswith("--- FAIL:")})
    if "panic:" in out and not fails: fails.append("PANIC")
    return ("pass" if r.returncode == 0 else "FAIL " + ",".join(fails)), out
