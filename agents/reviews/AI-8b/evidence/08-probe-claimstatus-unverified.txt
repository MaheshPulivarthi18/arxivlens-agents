package query

// QA AI-8b probes (not part of the PR). Prints claims for cases the PR tests don't pin literally.
import (
	"context"
	"encoding/json"
	"fmt"
	"math"
	"testing"
)

func TestQAProbeClaims(t *testing.T) {
	captureLog(t)
	ctx := context.Background()
	quote := "The Method section shows that the proposed method improves accuracy on every benchmark [1]."
	show := func(name string, resp *AskResponse) {
		var cl []string
		for _, c := range resp.Claims {
			m, _ := json.Marshal(c.Markers)
			cl = append(cl, fmt.Sprintf("%s%s text=%q", c.Status, m, c.Text[:min(len(c.Text), 20)]))
		}
		fmt.Printf("PROBE %-50s claims=%v citation_statuses=%v refusal=%q\n", name, cl, statuses(resp.Citations), resp.RefusalReason)
	}
	// P1: verbatim quote supported + paraphrase whose verifier call fails -> claim 2 has one unverified citation.
	r := &RAGService{verifier: &fakeVerifier{err: fmt.Errorf("down")}}
	show("P1 quote[1] + paraphrase[2], verifier down", r.finalize(ctx, "t", quote+" Training follows the described setup [2].", batchHits()))
	// P2: claim with citations [unsupported 0.2, unverified NaN] -> claimStatus rule says unverified.
	r = &RAGService{verifier: byPassage(map[string]float64{"Results": 0.2, "Discussion": math.NaN()})}
	show("P2 quote[1] + paraphrase[2][3] = [0.2, NaN]", r.finalize(ctx, "t", quote+" Training follows the described setup [2][3].", batchHits()))
	// P3: claim with citations [unsupported 0.2] only -> unsupported.
	r = &RAGService{verifier: byPassage(map[string]float64{"Results": 0.2})}
	show("P3 quote[1] + paraphrase[2] = [0.2]", r.finalize(ctx, "t", quote+" Training follows the described setup [2].", batchHits()))
	// P4: a fragment that is only a marker -> claim text after normQuote.
	r = &RAGService{verifier: simVerifier(constSim(0.9))}
	show("P4 quote[1] + trailing marker-only fragment '[2]'", r.finalize(ctx, "t", quote+" [2]", batchHits()))
}
