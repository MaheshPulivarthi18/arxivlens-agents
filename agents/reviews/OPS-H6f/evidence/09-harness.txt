import importlib.abc
import importlib.machinery
import sys
from unittest import mock

SKIP = {"simplejson", "_typeshed"}


class _Finder(importlib.abc.MetaPathFinder, importlib.abc.Loader):
    def find_spec(self, name, path, target=None):
        root = name.split(".")[0]
        if root in SKIP or name in sys.modules:
            return None
        for f in sys.meta_path:
            if f is self:
                continue
            try:
                if f.find_spec(name, path, target) is not None:
                    return None
            except Exception:
                pass
        return importlib.machinery.ModuleSpec(name, self, is_package=True)

    def create_module(self, spec):
        m = mock.MagicMock(name=spec.name)
        m.__path__, m.__spec__, m.__name__ = [], spec, spec.name
        return m

    def exec_module(self, module):
        pass


sys.meta_path.append(_Finder())

===== stub/psycopg/__init__.py
"""QA double of psycopg 3 (no psycopg / PyPI in the sandbox): a minimal PostgreSQL wire-protocol client over the Unix
socket (simple-query protocol, trust auth). One real server session per Connection, so row locks are real. Parameters
are interpolated client-side as SQL literals. autocommit=False: BEGIN is sent before the first statement, like psycopg."""
import os
import re
import socket
import struct
from urllib.parse import urlparse

from . import rows  # noqa: F401


class Error(Exception):
    def __init__(self, fields):
        self.fields = fields
        self.sqlstate = fields.get("C")
        super().__init__(f"{fields.get('S')}: {fields.get('M')} (SQLSTATE {fields.get('C')})")


OperationalError = DatabaseError = Error
_INT = {20, 21, 23, 26}
_FLOAT = {700, 701, 1700}


def _lit(v):
    if v is None:
        return "NULL"
    if isinstance(v, bool):
        return "TRUE" if v else "FALSE"
    if isinstance(v, (int, float)):
        return repr(v)
    if isinstance(v, (list, tuple)):
        items = ",".join('"' + str(x).replace("\\", "\\\\").replace('"', '\\"') + '"' for x in v)
        return "'{" + items.replace("'", "''") + "}'"
    return "'" + str(v).replace("'", "''") + "'"


def _interp(sql, params):
    if params is None:
        return sql
    it = iter(params)
    out = re.sub(r"%s", lambda m: _lit(next(it)), sql.replace("%%", "\x00"))
    rest = list(it)
    assert not rest, f"unused params {rest}"
    return out.replace("\x00", "%")


class Connection:
    def __init__(self, sockdir, port, user, db, autocommit):
        self.s = socket.socket(socket.AF_UNIX)
        self.s.connect(os.path.join(sockdir, f".s.PGSQL.{port}"))
        self.autocommit, self.in_tx, self.closed = autocommit, False, False
        body = struct.pack("!i", 196608) + b"user\0" + user.encode() + b"\0database\0" + db.encode() + b"\0\0"
        self.s.sendall(struct.pack("!i", len(body) + 4) + body)
        self._until_ready()

    def _recv(self, n):
        b = b""
        while len(b) < n:
            c = self.s.recv(n - len(b))
            if not c:
                raise Error({"M": "server closed the connection"})
            b += c
        return b

    def _msg(self):
        t = self._recv(1)
        ln = struct.unpack("!i", self._recv(4))[0]
        return t, self._recv(ln - 4)

    def _until_ready(self):
        results, cur, err = [], None, None
        while True:
            t, b = self._msg()
            if t == b"R":
                assert struct.unpack("!i", b[:4])[0] == 0, "only trust auth"
            elif t == b"T":
                n, off, cols = struct.unpack("!h", b[:2])[0], 2, []
                for _ in range(n):
                    end = b.index(b"\0", off)
                    name = b[off:end].decode()
                    oid = struct.unpack("!i", b[end + 7:end + 11])[0]
                    cols.append((name, oid))
                    off = end + 19
                cur = {"cols": cols, "rows": []}
            elif t == b"D":
                n, off, row = struct.unpack("!h", b[:2])[0], 2, []
                for name, oid in cur["cols"]:
                    ln = struct.unpack("!i", b[off:off + 4])[0]
                    off += 4
                    if ln < 0:
                        row.append(None)
                        continue
                    v = b[off:off + ln].decode()
                    off += ln
                    row.append(int(v) if oid in _INT else float(v) if oid in _FLOAT else (v == "t") if oid == 16 else v)
                cur["rows"].append(tuple(row))
            elif t == b"C":
                tag = b[:-1].decode()
                cnt = int(tag.split()[-1]) if tag.split()[-1].isdigit() else -1
                results.append((cur, cnt))
                cur = None
            elif t == b"E":
                f, off = {}, 0
                while b[off] != 0:
                    end = b.index(b"\0", off + 1)
                    f[chr(b[off])] = b[off + 1:end].decode()
                    off = end + 1
                err = Error(f)
            elif t == b"Z":
                self.status = b
                if err:
                    raise err
                return results

    def _simple(self, sql):
        q = sql.encode() + b"\0"
        self.s.sendall(b"Q" + struct.pack("!i", len(q) + 4) + q)
        return self._until_ready()

    def execute_sql(self, sql):
        if not self.autocommit and not self.in_tx:
            self._simple("BEGIN")
            self.in_tx = True
        return self._simple(sql)

    def cursor(self, row_factory=None):
        return Cursor(self, row_factory)

    def commit(self):
        if self.in_tx:
            self._simple("COMMIT")
        self.in_tx = False

    def rollback(self):
        if self.in_tx:
            self._simple("ROLLBACK")
        self.in_tx = False

    def close(self):
        if not self.closed:
            try:
                self.s.sendall(b"X" + struct.pack("!i", 4))
            finally:
                self.s.close()
                self.closed = True

    def __enter__(self):
        return self

    def __exit__(self, *a):
        self.close()


class Cursor:
    def __init__(self, conn, row_factory):
        self.conn, self.row_factory, self._rows, self.rowcount, self.description = conn, row_factory, [], -1, None

    def execute(self, sql, params=None):
        res = self.conn.execute_sql(_interp(sql, params))
        cur, cnt = res[-1] if res else (None, -1)
        self.rowcount = cnt
        if cur is None:
            self._rows, self.description = [], None
        else:
            names = [c[0] for c in cur["cols"]]
            self.description = names
            self._rows = [dict(zip(names, r)) for r in cur["rows"]] if self.row_factory else list(cur["rows"])
        return self

    def fetchone(self):
        return self._rows.pop(0) if self._rows else None

    def fetchall(self):
        r, self._rows = self._rows, []
        return r

    def close(self):
        pass

    def __enter__(self):
        return self

    def __exit__(self, *a):
        pass


def connect(url, autocommit=False):
    u = urlparse(url)
    return Connection(os.environ["QA_PGSOCKDIR"], u.port or 5432, u.username or "postgres", u.path.lstrip("/"), autocommit)

===== halfvec_shim.sql
-- QA shim: no pgvector in the sandbox. halfvec = text domain; a trigger normalises '[1.000000,...]' to pgvector's '[1,...]'
-- text form and enforces 768 dims, so the test's _vec() comparisons behave as on real pgvector.
CREATE DOMAIN halfvec AS text;
CREATE FUNCTION qa_norm_halfvec() RETURNS trigger LANGUAGE plpgsql AS $$
DECLARE parts text[];
BEGIN
  parts := string_to_array(btrim(NEW.embedding, '[] '), ',');
  IF array_length(parts, 1) IS DISTINCT FROM 768 THEN RAISE EXCEPTION 'expected 768 dimensions, not %', array_length(parts, 1); END IF;
  NEW.embedding := '[' || array_to_string(ARRAY(SELECT (x::real)::text FROM unnest(parts) WITH ORDINALITY AS u(x, n) ORDER BY n), ',') || ']';
  RETURN NEW;
END $$;
