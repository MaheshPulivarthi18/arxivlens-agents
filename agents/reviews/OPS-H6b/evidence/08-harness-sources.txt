#!/usr/bin/env python3
"""Single-edit mutants of internal/kafka/consumer.go (PR head) against the real franz-go v1.16.1 harness.
Each mutant: fresh copy of the harness (mktemp -d, never deleted), one str.replace that must match exactly once,
then go test twice: full package, and -run '^TestConsumerPollErrorAfterCancel$'. Build failure -> INVALID (never KILLED)."""
import os, re, shutil, subprocess, tempfile
H = os.path.join(os.path.dirname(os.path.abspath(__file__)), "..", "harness")
F = "internal/kafka/consumer.go"
G = "\t\t\tif ctx.Err() != nil && errors.Is(err, ctx.Err()) {\n\t\t\t\treturn\n\t\t\t}\n"
CL = "\t\t\tclosed = closed || errors.Is(err, kgo.ErrClientClosed)\n"
M = [
 ("M1 (AC4a) remove the guard", G, ""),
 ("M2 (AC4b) guard on ctx.Err() != nil alone", "if ctx.Err() != nil && errors.Is(err, ctx.Err()) {", "if ctx.Err() != nil {"),
 ("M3 (AC4c) skip-all-after-cancel before closed", CL + G, "\t\t\tif ctx.Err() != nil {\n\t\t\t\treturn\n\t\t\t}\n" + CL),
 ("M4 && -> ||", "if ctx.Err() != nil && errors.Is(err, ctx.Err()) {", "if ctx.Err() != nil || errors.Is(err, ctx.Err()) {"),
 ("M5 negate errors.Is", "if ctx.Err() != nil && errors.Is(err, ctx.Err()) {", "if ctx.Err() != nil && !errors.Is(err, ctx.Err()) {"),
 ("M6 full guard moved before closed (keeps errors.Is)", CL + G, G + CL),
 ("M7 drop ctx.Err() != nil operand (expect equivalent)", "if ctx.Err() != nil && errors.Is(err, ctx.Err()) {", "if errors.Is(err, ctx.Err()) {"),
 ("M8 errors.Is -> == (expect equivalent)", "errors.Is(err, ctx.Err()) {", "err == ctx.Err() {"),
 ("M9 ctx.Err() -> context.Canceled (expect equivalent here)", "errors.Is(err, ctx.Err()) {", "errors.Is(err, context.Canceled) {"),
 ("M10 drop closed computation", CL, "\t\t\t_ = kgo.ErrClientClosed\n"),
]
def run(d, extra):
    p = subprocess.run(["go", "test", "-count=1", *extra, "./internal/kafka/"], cwd=d, capture_output=True, text=True,
                       env={**os.environ, "GOFLAGS": "-mod=mod", "GOPROXY": "off", "GOTOOLCHAIN": "local"})
    out = p.stdout + p.stderr
    if "[build failed]" in out or "[setup failed]" in out: return "INVALID", out.splitlines()[:3]
    if p.returncode == 0: return "SURVIVED", []
    return "KILLED", sorted(set(re.findall(r"--- FAIL: (\S+)", out)))
print(f"{'mutant':55} {'full pkg':9} {'-run AfterCancel':17} failing tests (full pkg)")
for name, old, new in M:
    d = tempfile.mkdtemp(prefix="mut-")
    shutil.copytree(H, d, dirs_exist_ok=True, symlinks=True)
    # harness go.mod uses ../ relative replaces; rewrite to absolute
    gm = os.path.join(d, "go.mod"); s = open(gm).read()

    open(gm, "w").write(s)
    src = open(os.path.join(d, F)).read()
    n = src.count(old)
    if n != 1: print(f"{name:55} BAD-PATTERN (matches={n})"); continue

    v1, f1 = run(d, [])

    print(f"{name:55} {v1:9} {v2:17} {' '.join(f1)}")
