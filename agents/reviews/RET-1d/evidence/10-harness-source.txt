#!/usr/bin/env bash
# For each mutant: fresh module copy of the PR head, apply one anchored perl substitution to server.go,
# check exactly the intended code changed, then run the AC6 command. Build failure => INVALID.
H=/tmp/claude-0/-home-claude/5e719a67-b74f-583a-8173-6f00308d6e83/scratchpad/qa-RET-1d-r1
export GOFLAGS=-mod=mod GOPROXY=off GONOSUMDB='*' GOTOOLCHAIN=local
SRC=${SRC:-/root/wt/qa-RET-1d}
printf '%-3s %-9s %s\n' id verdict mutant
while IFS='|' read -r id desc expr; do
  d=$(mktemp -d "$H/mut-$id-XXXX"); $H/bin/mkmod.sh "$SRC" "$d"
  f=$d/internal/query/server.go
  perl -0pi -e "$expr" "$f"
  if diff -q "$SRC/internal/query/server.go" "$f" >/dev/null; then printf '%-3s %-9s %s\n' "$id" NOT-APPLIED "$desc"; continue; fi
  diff "$SRC/internal/query/server.go" "$f" > "$d/mutant.diff"
  if ! (cd "$d" && go vet ./internal/query/ >"$d/build.log" 2>&1); then printf '%-3s %-9s %s\n' "$id" INVALID "$desc"; continue; fi
  (cd "$d" && go test -count=1 -run '^(TestWriteJSON|TestHandlersEncodeContract)$' ./internal/query/ >"$d/test.log" 2>&1); rc=$?
  if [ $rc -ne 0 ]; then v=KILLED; why=$(grep -m1 -- '--- FAIL: TestHandlersEncodeContract/' "$d/test.log" | sed 's/ (.*//;s/.*\///'); else v=SURVIVED; why=""; fi
  printf '%-3s %-9s %s%s\n' "$id" "$v" "$desc" "${why:+  [first failing row: $why]}"
done < $H/bin/mutants.txt
