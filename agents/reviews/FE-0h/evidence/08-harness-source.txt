"""AC2: WCAG 2.x contrast of --input (and --border for reference) vs each call site's fill and parent, parsed from index.css."""
import re, sys
css = open(sys.argv[1]).read()
def block(sel):
    m = re.search(re.escape(sel) + r'\s*\{(.*?)\n  \}', css, re.S)
    return dict(re.findall(r'--([\w-]+):\s*([\d.]+ [\d.]+% [\d.]+%);', m.group(1)))
themes = {'light': block(':root'), 'dark': block('.dark')}
def hsl2rgb(s):
    h, sat, l = s.split(); h = float(h)/360; sat = float(sat[:-1])/100; l = float(l[:-1])/100
    def f(n):
        k = (n + h*12) % 12; a = sat*min(l, 1-l)
        return l - a*max(-1, min(k-3, 9-k, 1))
    return [f(0), f(8), f(4)]
def lum(rgb):
    c = [x/12.92 if x <= 0.04045 else ((x+0.055)/1.055)**2.4 for x in rgb]
    return 0.2126*c[0] + 0.7152*c[1] + 0.0722*c[2]
def ratio(a, b):
    la, lb = sorted([lum(hsl2rgb(a)), lum(hsl2rgb(b))], reverse=True)
    return (la+0.05)/(lb+0.05)
sites = [('IngestModal arXiv ID input', 'background', 'fill'), ('IngestModal arXiv ID input', 'card', 'parent (DialogContent bg-card)'),
         ('PapersLibrary filter input', 'card', 'fill'), ('PapersLibrary filter input', 'background', 'parent (App bg-background)')]
ok = True
print(f"{'theme':6} {'call site':28} {'surface':32} {'--input':>8} {'--border':>9}  >=3:1")
for t, v in themes.items():
    print(f"# {t}: --input = {v['input']}, --border = {v['border']}, --background = {v['background']}, --card = {v['card']}")
    for site, surf, role in sites:

// Minimal Vitest shim (logic check only; CI Vitest is the pass/fail). Prints one line per test.
const tests = []; let prefix = []
export function describe(name, fn) { prefix.push(name); fn(); prefix.pop() }
export function it(name, fn) { tests.push([[...prefix, name].join(' > '), fn]) }
it.each = (rows) => (name, fn) => rows.forEach((r) => it(name.replace('%s', r), () => fn(r)))
const fmt = (v) => JSON.stringify(v)
function matchers(actual, msg, neg) {
  const chk = (pass, text) => { if (pass === neg) throw new Error((msg ? msg + ': ' : '') + (neg ? 'not ' : '') + text) }
  return {
    toMatch: (re) => chk(re.test(actual), `expected ${fmt(actual)} to match ${re}`),
    toBeGreaterThan: (n) => chk(actual > n, `expected ${actual} > ${n}`),
    toBeNull: () => chk(actual === null, `expected ${fmt(actual)} to be null`),
    toEqual: (e) => chk(fmt(actual) === fmt(e), `expected ${fmt(actual)} to equal ${fmt(e)}`),
  }
}
export function expect(actual, msg) { const m = matchers(actual, msg, false); m.not = matchers(actual, msg, true); return m }
setTimeout(() => {
  let fail = 0
  for (const [n, fn] of tests) { try { fn(); console.log('  PASS ' + n) } catch (e) { fail++; console.log('  FAIL ' + n + '  -> ' + e.message) } }
  console.log(`  Tests: ${tests.length - fail} passed, ${fail} failed (${tests.length})`)
  process.exitCode = fail ? 1 : 0
}, 0)
