def run(tree):
    p = subprocess.run([sys.executable, "-m", "pytest", "-q", "-x", "-p", "no:cacheprovider", "tests/test_evals_data.py"],
                       cwd=tree, capture_output=True, text=True, timeout=300)
    last = [ln for ln in p.stdout.splitlines() if ln.strip()][-1:] or ["?"]
    failed = [ln.split(" ")[1] for ln in p.stdout.splitlines() if ln.startswith("FAILED ")]
    return p.returncode, last[0], failed

rows.append(("control (unmutated PR head)", "PASS" if rc == 0 else f"BROKEN rc={rc}", last, []))
for i, (name, path, old, new) in enumerate(MUTANTS):
    tree = os.path.join(WORK, f"mut{i:02d}")
    shutil.copytree(base_tree, tree)

    src = open(f, encoding="utf-8").read()
    n = src.count(old)
    if n != 1:
        rows.append((name, f"INVALID (anchor found {n}x)", "", []))

    open(f, "w", encoding="utf-8").write(src.replace(old, new))

    diffs.append((name, [ln for ln in d.splitlines() if ln[:1] in "+-" and not ln.startswith(("+++", "---"))]))
    rc, last, failed = run(tree)
    rows.append((name, "KILLED" if rc == 1 else ("SURVIVED" if rc == 0 else f"ERROR rc={rc}"), last, failed))
