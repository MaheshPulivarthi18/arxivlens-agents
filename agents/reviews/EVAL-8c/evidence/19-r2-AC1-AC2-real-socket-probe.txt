"""QA EVAL-8c: drive the PR's real eval_gate.main against a real local HTTP server (no urllib monkeypatch).

Only the stack-facing pieces are stubbed (embedder_device, retrieval, /v1/ask). The Ollama probe goes over a real

class H(http.server.BaseHTTPRequestHandler):
    def log_message(self, *a):
        pass

    def do_GET(self):
        SEEN.append((self.path, self.headers.get("Authorization")))
        m = MODE["m"]
        if m == "slow":
            time.sleep(7)
        if m == "401":
            self.send_response(401); self.end_headers(); self.wfile.write(b"no " + SECRET.encode()); return
        if m == "nonjson":
            body = b"<html>" + SECRET.encode()
        elif self.path == "/api/version":
            body = b'{"version":"0.40.16"}'
        elif self.path == "/api/tags":
            models = [{"name": "other:1b", "digest": "f" * 64}]
            if m != "missing":
                models.append({"name": "qwen2.5:3b-instruct", "model": "qwen2.5:3b-instruct", "digest": DIGEST})
            body = json.dumps({"models": models}).encode()
        else:
            self.send_response(404); self.end_headers(); return
        self.send_response(200); self.send_header("Content-Type", "application/json"); self.end_headers()
        self.wfile.write(body)
