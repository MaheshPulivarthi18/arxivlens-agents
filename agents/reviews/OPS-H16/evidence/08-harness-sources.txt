#!/bin/bash
# Real npm (no registry needed): PR-head vs origin/main recipe, on scratch frontend/ dirs.
S=/tmp/claude-0/-home-claude/a2b9483d-cd17-536a-9d28-9b2b9b15c268/scratchpad/qa-OPS-H16-r1
HEAD_MK=/root/wt/qa-OPS-H16/Makefile; OLD_MK=$S/oldmk/Makefile
echo "npm $(npm --version), node $(node --version); PR head $(git -C /root/wt/qa-OPS-H16 rev-parse --short HEAD)"
mk() { d=$(mktemp -d "$S/proj.XXXX"); mkdir "$d/frontend"; printf '%s' "$1" > "$d/frontend/package.json"; echo "$d"; }
NOTEST='{"name":"frontend","private":true,"scripts":{"dev":"vite","build":"vite build","lint":"oxlint"}}'
PASS='{"name":"frontend","private":true,"scripts":{"dev":"vite","build":"vite build","lint":"oxlint","test":"echo FAKE-VITEST-OK"}}'
FAIL='{"name":"frontend","private":true,"scripts":{"dev":"vite","build":"vite build","lint":"oxlint","test":"echo FAKE-VITEST-FAIL; exit 1"}}'
for case in "NOTEST:$NOTEST" "PASS:$PASS" "FAIL:$FAIL"; do
  name=${case%%:*}; json=${case#*:}
  for mkf in HEAD:$HEAD_MK OLD:$OLD_MK; do
    d=$(mk "$json"); echo "=== recipe=${mkf%%:*} package.json test script: $name"
    (cd "$d" && make -f "${mkf#*:}" test-fe 2>&1); rc=$?
    echo "--> make exit=$rc"; echo
  done
done

[exit code 0]
