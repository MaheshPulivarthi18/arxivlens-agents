def run_tests(root, runs):
    env = dict(os.environ, PYTHONPATH=f"{H}/stubs:{H}:{root}")
    fails, first_fail = 0, ""
    for _ in range(runs):
        p = subprocess.run([sys.executable, "-m", "pytest", "-p", "qa_stubs", "-p", "no:cacheprovider", "-q", "-rf",
                            "tests/test_verify.py", "-k", SEL], cwd=root, env=env, capture_output=True, text=True)
        if p.returncode != 0:
            fails += 1
            if not first_fail:
                first_fail = "; ".join(l[7:].split(" - ")[0].split("::")[-1] + (" - " + l.split(" - ", 1)[1][:90] if " - " in l else "")
                                       for l in p.stdout.splitlines() if l.startswith(("FAILED", "ERROR")))
    return fails, first_fail

Usage: mutate_lock.py <app_dir> <harness_dir>"""
import os, shutil, subprocess, sys, tempfile, re
APP, H = sys.argv[1], sys.argv[2]
API = "services/embedder/api.py"
LOCKED = "    with _ingest_lock:\n        return _ingest_paper(req)"

 ("M2", [(LOCKED, "    if True:\n        return _ingest_paper(req)")], "no lock at all"),
 ("M13", [(LOCKED, "    if True:\n        return _ingest_paper(req)"),
          ("        processor, embedder_worker = _ingest_workers()\n",
           "        with _ingest_lock:\n            processor, embedder_worker = _ingest_workers()\n")],
  "lock only around worker construction, pipeline unlocked"),
]
for mid, edits, desc in M:
    root = tempfile.mkdtemp(prefix=f"qa-ai12-{mid}-")
    for d in ("services", "scripts", "shared", "tests"):
        shutil.copytree(os.path.join(APP, d), os.path.join(root, d))
    p = os.path.join(root, API); s = open(p).read()
    for o, n in edits:
        assert s.count(o) == 1, (mid, o); s = s.replace(o, n)
    open(p, "w").write(s)
    env = dict(os.environ, PYTHONPATH=f"{H}/stubs:{H}:{root}")
    tally = {}
    for _ in range(20):
        r = subprocess.run([sys.executable, "-m", "pytest", "-p", "qa_stubs", "-p", "no:cacheprovider", "-q",
                            "tests/test_verify.py", "-k", "concurrent"], cwd=root, env=env, capture_output=True, text=True)
        diff = re.findall(r"^E\s+(\{'\w+': \S+\} != \{'\w+': \S+\})", r.stdout, re.M)
        key = ("pass" if r.returncode == 0 else "FAIL, differing: " + (", ".join(diff) if diff else "?"))
        tally[key] = tally.get(key, 0) + 1
    print(f"{mid} ({desc}) 20 runs:")
    for k, v in tally.items():
        print(f"   {v:2}x {k}")

"""pytest plugin: any module that can't be imported becomes a MagicMock module (torch, transformers, kafka, minio...)."""
import importlib.abc
import importlib.machinery
import sys
from unittest import mock

REAL_ONLY = {"simplejson", "_pytest", "pytest", "pydantic", "numpy", "fastapi", "psycopg", "ulid"}


class _Loader(importlib.abc.Loader):
    def create_module(self, spec):
        m = mock.MagicMock(name=spec.name)
        m.__path__ = []
        m.__spec__ = spec
        m.__name__ = spec.name
        return m

    def exec_module(self, module):
        pass


class _Finder(importlib.abc.MetaPathFinder):
    def find_spec(self, name, path, target=None):
        if name.split(".")[0] in REAL_ONLY or name.startswith(("services", "scripts", "shared", "tests")):
            return None
        for f in sys.meta_path:
            if f is self:
                continue
            try:
                s = f.find_spec(name, path, target)
            except Exception:
                s = None
            if s is not None:
                return None
        return importlib.machinery.ModuleSpec(name, _Loader(), is_package=True)


sys.meta_path.append(_Finder())
