func TestQAProbeSurvivors(t *testing.T) {
	ctx := context.Background()
	d, err := Connect(ctx, "stub")
	if err != nil {
		t.Fatal(err)
	}
	if err := d.RunMigrations(ctx, "../../migrations/000005_crawl_watermarks.up.sql"); err != nil {
		t.Fatal(err)
	}
	cancelled, cancel := context.WithCancel(ctx)
	cancel()
	_, ok, err := d.Watermark(cancelled, "qa.probe")
	t.Logf("M6 probe: Watermark(cancelled ctx) -> ok=%v err=%v  (want err != nil)", ok, err)
	err = d.AdvanceWatermark(cancelled, "qa.probe", time.Now())
	t.Logf("M8 probe: AdvanceWatermark(cancelled ctx) -> err=%v  (want err != nil)", err)
	t0 := time.Date(2026, 1, 1, 0, 0, 0, 0, time.UTC)
	_ = d.AdvanceWatermark(ctx, "qa.probe", t0)
	var u1, u2 time.Time
	_ = d.Pool.QueryRow(ctx, `SELECT updated_at FROM crawl_watermarks WHERE category = $1`, "qa.probe").Scan(&u1)
	time.Sleep(50 * time.Millisecond)
	_ = d.AdvanceWatermark(ctx, "qa.probe", t0.Add(time.Hour))
	_ = d.Pool.QueryRow(ctx, `SELECT updated_at FROM crawl_watermarks WHERE category = $1`, "qa.probe").Scan(&u2)
	t.Logf("M7 probe: updated_at bumped on conflict update: %v  (want true)", u2.After(u1))
}

// QA STUB (not real pgxpool): executes each statement with `psql -At -v ON_ERROR_STOP=1` against $QA_PSQL_HOST:$QA_PSQL_PORT,
// substituting $N placeholders with SQL literals. After Close every call returns "closed pool" (as pgxpool does).
package pgxpool

import (
	"context"
	"errors"
	"fmt"
	"os"
	"os/exec"
	"regexp"
	"strconv"
	"strings"
	"time"

	"github.com/jackc/pgx/v5"
	"github.com/jackc/pgx/v5/pgconn"
)

type Config struct {
	MaxConns, MinConns int32
	MaxConnLifetime    time.Duration
	dsn                string
}

func ParseConfig(s string) (*Config, error) { return &Config{dsn: s}, nil }

type Pool struct{ closed bool }

func NewWithConfig(ctx context.Context, c *Config) (*Pool, error) { return &Pool{}, nil }

var errClosed = errors.New("closed pool")

func (p *Pool) Close() { p.closed = true }

func (p *Pool) Ping(ctx context.Context) error { _, err := p.run(ctx, "SELECT 1"); return err }

var ph = regexp.MustCompile(`\$([0-9]+)`)

func lit(a any) string {
	switch v := a.(type) {
	case string:
		return "'" + strings.ReplaceAll(v, "'", "''") + "'"
	case time.Time:
		return "'" + v.UTC().Format(time.RFC3339Nano) + "'::timestamptz"
	case int, int32, int64:
		return fmt.Sprint(v)
	case nil:
		return "NULL"
	}
	panic(fmt.Sprintf("stub: unsupported arg %T", a))
}

func (p *Pool) run(ctx context.Context, sql string, args ...any) (string, error) {
	if p.closed {
		return "", errClosed
	}
	if err := ctx.Err(); err != nil {
		return "", err
	}
	sql = ph.ReplaceAllStringFunc(sql, func(m string) string { n, _ := strconv.Atoi(m[1:]); return lit(args[n-1]) })
	cmd := exec.Command("psql", "-h", os.Getenv("QA_PSQL_HOST"), "-p", os.Getenv("QA_PSQL_PORT"), "-U", "postgres", "-d", "qa",
		"-X", "-q", "-At", "-v", "ON_ERROR_STOP=1", "-c", "SET TIME ZONE 'UTC'", "-c", sql)
	out, err := cmd.CombinedOutput()
	if err != nil {
		return "", fmt.Errorf("psql: %v: %s", err, strings.TrimSpace(string(out)))
	}
	return strings.TrimRight(string(out), "\n"), nil
}

func (p *Pool) Exec(ctx context.Context, sql string, args ...any) (pgconn.CommandTag, error) {
	_, err := p.run(ctx, sql, args...)
	return pgconn.CommandTag{}, err
}

type row struct {
	out string
	err error
}

func (r row) Scan(dest ...any) error {
	if r.err != nil {
		return r.err
	}
	if r.out == "" {
		return pgx.ErrNoRows
	}
	f := strings.Split(strings.SplitN(r.out, "\n", 2)[0], "|")
	for i, d := range dest {
		switch v := d.(type) {
		case *time.Time:
			t, err := time.Parse("2006-01-02 15:04:05.999999-07", f[i])
			if err != nil {
				return err
			}
			*v = t
		case *int64:
			n, err := strconv.ParseInt(f[i], 10, 64)
			if err != nil {
				return err
			}
			*v = n
		case *string:
			*v = f[i]
		default:
			return fmt.Errorf("stub: unsupported scan dest %T", d)
		}
	}
	return nil
}

func (p *Pool) QueryRow(ctx context.Context, sql string, args ...any) pgx.Row {
	out, err := p.run(ctx, sql, args...)
	return row{out, err}
}

type rows struct{}

func (rows) Next() bool          { return false }
func (rows) Scan(...any) error   { return errors.New("stub: Query unsupported") }
func (rows) Err() error          { return nil }
func (rows) Close()              {}

func (p *Pool) Query(ctx context.Context, sql string, args ...any) (pgx.Rows, error) {
	return rows{}, errors.New("stub: Query unsupported")
}
