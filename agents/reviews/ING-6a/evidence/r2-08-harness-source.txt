#!/bin/bash
# mutate2.sh <app tree> <mutants file>: like mutate.sh, but prints EVERY failing (sub)test of the full run. Build failure = INVALID.
W=$1; MF=$2; H=$(cd "$(dirname "$0")" && pwd); S=$(dirname $H)
export GOTOOLCHAIN=local GOPROXY=off GOFLAGS=-mod=mod
python3 -I - "$W" "$H" "$S" "$MF" <<'PY'
import subprocess, sys, tempfile, os
W, H, S, MF = sys.argv[1:5]
MUTDIR = H
exec(open(MF).read())
def run(m, args):
    r = subprocess.run(["go", "test", "-count=1", "-timeout", "30s"] + args + ["./internal/crawler/"], cwd=m, capture_output=True, text=True)
    out = r.stdout + r.stderr
    if "[build failed]" in out or "[setup failed]" in out: return "INVALID(build)", ""
    if r.returncode == 0: return "SURVIVED", ""
    fails = [l.strip().split()[2] for l in out.splitlines() if l.strip().startswith("--- FAIL") and "/" in l]
    fails = fails or [l.strip().split()[2] for l in out.splitlines() if l.strip().startswith("--- FAIL")]
    if "panic: test timed out" in out: fails.append("TIMEOUT")
    return "KILLED", " ; ".join(f.split("/",1)[-1] for f in fails)
rows = []
for name, test, old, new in M:
    m = tempfile.mkdtemp(dir=S, prefix="mut2.")
    subprocess.run([os.path.join(H, "build.sh"), W, m], check=True)
    p = os.path.join(m, "internal/crawler/loop.go"); src = open(p).read()
    if src.count(old) != 1: rows.append((name, "NOT APPLIED", f"pattern count {src.count(old)}", "", "")); continue
    open(p, "w").write(src.replace(old, new))
    a, fa = run(m, []); b, _ = run(m, ["-run", f"^{test}$"])
    rows.append((name, a, b, test, fa))
k = sum(1 for r in rows if r[1] == "KILLED" and r[2] == "KILLED")
print(f"SUMMARY: {k}/{len(rows)} killed in BOTH the full package and -run '^T$'")
print(f"{'mutant':44} | full pkg | -run ^T$ | T")
for r in rows: print(f"{r[0]:44} | {r[1]:8} | {r[2]:8} | {r[3]}")
print("\nfailing (sub)tests, full package run:")
for r in rows: print(f"  {r[0].split()[0]}: {r[4]}")
PY
