// Stub of github.com/oklog/ulid/v2 (signatures used by shared/events only).
package ulid

import (
	"fmt"
	"io"
	"time"
)

type ULID [16]byte

func (u ULID) String() string                       { return fmt.Sprintf("%x", u[:]) }
func Monotonic(r io.Reader, inc uint64) io.Reader   { return r }
func Timestamp(t time.Time) uint64                  { return uint64(t.UnixMilli()) }
func MustNew(ms uint64, r io.Reader) ULID {
	var u ULID
	for i := 0; i < 6; i++ { u[i] = byte(ms >> (8 * (5 - i))) }
	_, _ = io.ReadFull(r, u[6:])
	return u
}

#!/bin/bash
# mutate.sh <app tree> : for each mutant, fresh scratch module, apply, run full crawler pkg and -run '^T$'. Build failure = INVALID.
W=$1; H=$(cd "$(dirname "$0")" && pwd); S=$(dirname $H)
export GOTOOLCHAIN=local GOPROXY=off GOFLAGS=-mod=mod
python3 -I - "$W" "$H" "$S" <<'PY'
import subprocess, sys, tempfile, os
W, H, S = sys.argv[1:4]
exec(open(os.path.join(H, "mutants.py")).read())
def run(m, args):
    r = subprocess.run(["go", "test", "-count=1", "-timeout", "30s"] + args + ["./internal/crawler/"], cwd=m, capture_output=True, text=True)
    out = r.stdout + r.stderr
    if "[build failed]" in out or "[setup failed]" in out: return "INVALID(build)", ""
    if r.returncode == 0: return "SURVIVED", ""
    fails = [l.strip().split()[2] for l in out.splitlines() if l.strip().startswith("--- FAIL")]
    if "panic: test timed out" in out: fails.append("TIMEOUT")
    return "KILLED", ",".join(fails[:3])
print(f"{'mutant':48} | full pkg | -run ^T$ | failing tests (full)")
for name, test, old, new in M:
    m = tempfile.mkdtemp(dir=S, prefix="mut.")
    subprocess.run([os.path.join(H, "build.sh"), W, m], check=True)
    p = os.path.join(m, "internal/crawler/loop.go"); src = open(p).read()
    if src.count(old) != 1: print(f"{name:48} | NOT APPLIED (pattern count {src.count(old)})"); continue
    open(p, "w").write(src.replace(old, new))
    a, fa = run(m, []); b, _ = run(m, ["-run", f"^{test}$"])
    print(f"{name:48} | {a:8} | {b:8} | {fa}")
PY

package crawler

// QA probe (not in the PR): undated entries on a FIRST run and an undated entry at the end of a page.
import (
	"context"
	"strings"
	"testing"
	"time"
)

func TestQAProbeUndated(t *testing.T) {
	// first run, limit 3: design §2 says an undated entry is "neither a candidate nor a boundary"
	f := &fakeFeed{entries: feed(15, -1, 14)}
	got, _, _ := walk(context.Background(), f, "cs.LG", time.Time{}, false, 3, 100)
	t.Logf("first run feed(15,undated,14) limit 3 -> candidates %q", ids(got))
	if strings.Contains(ids(got), "undated") {
		t.Errorf("undated entry became a first-run candidate: %q", ids(got))
	}
	// with W: page [15, 9, undated] crosses W at its last DATED entry -> must stop after 1 request
	w := t0.Add(10 * time.Minute)
	f = &fakeFeed{entries: feed(15, 9, -1, 8, 7, 6)}
	got, st, _ := walk(context.Background(), f, "cs.LG", w, true, 3, 100)
	t.Logf("W=10 feed(15,9,undated,8,7,6) page 3 -> requests %q candidates %q stats %+v", strings.Join(f.calls, " "), ids(got), st)
	if len(f.calls) != 1 {
		t.Errorf("walk did not stop at the page whose last dated entry is <= W: requests %q", strings.Join(f.calls, " "))
	}
	// out_of_order across an undated entry: 9 then undated then 13 is out of order
	f = &fakeFeed{entries: feed(15, 9, -1, 13, 12, 11, 5, 4, 3)}
	_, st, _ = walk(context.Background(), f, "cs.LG", time.Time{}, false, 9, 100)
	t.Logf("first run feed(15,9,undated,13,...) -> out_of_order=%d", st.OutOfOrder)
	if st.OutOfOrder != 1 {
		t.Errorf("out_of_order = %d across an undated entry, want 1", st.OutOfOrder)
	}
}
