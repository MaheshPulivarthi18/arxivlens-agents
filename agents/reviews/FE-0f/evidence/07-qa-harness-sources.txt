"""FE-0f AC2: contrast of Button variant="clinical" text on its fill, idle + hover,
light + dark, token values parsed from frontend/src/index.css (cwd = app at PR head).
Functions = FE-0 design appendix (exact HSL->sRGB, WCAG 2.x relative luminance).
Hover fill hover:bg-primary/90 = primary at alpha 0.9 composited (sRGB, as browsers do) over the surface."""
import colorsys, re, sys
css = open("frontend/src/index.css").read()
def block(sel):
    m = re.search(re.escape(sel) + r"\s*\{(.*?)\n\s*\}", css, re.S); return m.group(1)
def parse(b): return dict(re.findall(r"--([\w-]+):\s*([^;]+);", b))
T = {"light": parse(block(":root")), "dark": {**parse(block(":root")), **parse(block(".dark"))}}
def rgb(h):
    H, S, L = [float(x.rstrip('%')) for x in h.split()]
    return colorsys.hls_to_rgb(H / 360, L / 100, S / 100)
def lumc(c):
    f = lambda v: v / 12.92 if v <= 0.04045 else ((v + 0.055) / 1.055) ** 2.4
    r, g, b = map(f, c); return 0.2126 * r + 0.7152 * g + 0.0722 * b
def ratio_c(a, b):
    la, lb = lumc(a), lumc(b); return (max(la, lb) + 0.05) / (min(la, lb) + 0.05)
def over(fg, a, bg): return tuple(a * x + (1 - a) * y for x, y in zip(fg, bg))
# old variant, for the before/after: white on emerald-600 #059669 / teal-600 #0d9488
hexc = lambda h: tuple(int(h[i:i+2], 16) / 255 for i in (1, 3, 5))
print("Source: frontend/src/index.css   classes: " + re.search(r'clinical:\s*\n\s*"([^"]+)"', open("frontend/src/components/ui/button.jsx").read()).group(1))
for t in ("light", "dark"):
    print(f"[{t}] primary={T[t]['primary']!r} primary-foreground={T[t]['primary-foreground']!r} background={T[t]['background']!r} card={T[t]['card']!r} supported={T[t]['supported']!r}")
print()
print(f"{'state':<42}{'light':>9}{'dark':>9}  result (min 4.5:1, WCAG 1.4.3)")
rows = [("idle: fg on primary", None)] + [(f"hover: fg on primary/90 over {s}", s) for s in ("background", "card", "supported")]
ok = True
for label, surf in rows:
    vals = []
    for t in ("light", "dark"):
        fg, p = rgb(T[t]["primary-foreground"]), rgb(T[t]["primary"])
        fill = p if surf is None else over(p, 0.9, rgb(T[t][surf]))
        vals.append(ratio_c(fg, fill))
    res = "PASS" if min(vals) >= 4.5 else "FAIL"; ok &= res == "PASS"
    print(f"{label:<42}{vals[0]:>8.2f}:1{vals[1]:>7.2f}:1  {res}")
print()
print("before (main): white on emerald-600 / teal-600 gradient ends (theme-independent)")
for n, h in (("emerald-600 #059669", "#059669"), ("teal-600 #0d9488", "#0d9488")):
    print(f"  white on {n:<20} {ratio_c((1,1,1), hexc(h)):.2f}:1")
print("\nobservation (not an AC2 pair): HeroSearch.jsx:81 loading spinner is border-t-white (caller class, out of scope) on the disabled (opacity-50) button")
for t in ("light", "dark"):
    print(f"  [{t}] white on primary {ratio_c((1,1,1), rgb(T[t]['primary'])):.2f}:1   (vs primary-foreground {ratio_c(rgb(T[t]['primary-foreground']), rgb(T[t]['primary'])):.2f}:1)")
print("\nfocus ring (base class, unchanged): ring token == primary?", {t: T[t]['ring'] == T[t]['primary'] for t in T})
print("ALL AC2 PAIRS PASS" if ok else "AC2 FAIL"); sys.exit(0 if ok else 1)
