// Build the app's CSS with Tailwind v4's compile() in v3-compat mode (@config = the PR's tailwind.config.js minus @tailwindcss/typography,

const cfgSrc = fs.readFileSync(`${FE}/tailwind.config.js`, 'utf8').replace(/^import typography.*$/m, '').replace(/plugins:\s*\[typography\]/, 'plugins: []');
const cfgPath = `${path.dirname(OUT)}/qa.tailwind.config.mjs`; fs.writeFileSync(cfgPath, cfgSrc);

const c = await compile(css, {
  base: FE,
  loadStylesheet: async (id, base) => { const p = id === 'tailwindcss' ? `${TW}/index.css` : path.resolve(base, id); return { path: p, base: path.dirname(p), content: fs.readFileSync(p, 'utf8') }; },
  loadModule: async (id, base) => { const p = path.resolve(base, id); return { path: p, base: path.dirname(p), module: (await import(p)).default }; },
});
const files = []; const walk = (d) => { for (const f of fs.readdirSync(d)) { const p = `${d}/${f}`; if (fs.statSync(p).isDirectory()) walk(p); else if (/\.(jsx?|html)$/.test(f) && !/\.test\./.test(f)) files.push(p); } };

===== stubs-cva.js
// QA stand-in for class-variance-authority's cva(base, {variants, defaultVariants}) (same lookup semantics, no compoundVariants).
export function cva(base, cfg = {}) {
  return (props = {}) => {
    const out = [base];
    for (const [k, map] of Object.entries(cfg.variants || {})) {
      const v = props[k] ?? (cfg.defaultVariants || {})[k];
      if (v != null && map[v]) out.push(map[v]);
    }
    out.push(props.className, props.class);
    return out.filter(Boolean).join(' ');
  };
}
