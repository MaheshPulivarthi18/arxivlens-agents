#!/bin/bash
# usage: mutants.sh <pristine frontend dir> — one single-edit mutant per r2 test target; each in a fresh mktemp copy.
H=$(dirname "$0"); SRC=$1
run() { # id file perl-expr description
  local id=$1 f=$2 expr=$3 desc=$4 d; d=$(mktemp -d "$H/../m-XXXX"); cp -r "$SRC/src" "$d/src"
  perl -0pi -e "$expr" "$d/src/$f"
  local delta; delta=$(diff "$SRC/src/$f" "$d/src/$f" | grep '^[<>]' | sed 's/^/      /')
  if [ -z "$delta" ]; then echo "$id INVALID (substitution did not apply) — $desc"; return; fi
  local res; res=$(node "$H/run.mjs" "$d" 2>&1)
  local fails; fails=$(echo "$res" | grep -E '^(FAIL|PAGEERROR)' | sed 's/^/      /')
  if ! echo "$res" | grep -qE '^(pass|FAIL) '; then echo "$id INVALID (harness did not run) — $desc"; echo "$res" | head -3 | sed 's/^/      /';
  elif [ -n "$fails" ]; then echo "$id KILLED — $desc"; echo "$fails"; else echo "$id SURVIVED — $desc"; fi
  echo "$delta"
}
