# QA EVAL-6c base control: the PR's tests/test_evals_data.py run against origin/main's scripts/evaluate_answers.py,
# plus collected test counts main file vs head file.  argv: <app checkout> <head rev>
set -u; APP=$1; REV=$2
B=$(mktemp -d); git -C $APP archive origin/main | tar -x -C $B
git -C $APP show $REV:tests/test_evals_data.py > $B/tests/test_evals_data.py
H=$(mktemp -d); git -C $APP archive $REV | tar -x -C $H
M=$(mktemp -d); git -C $APP archive origin/main | tar -x -C $M
echo "== collected test ids in tests/test_evals_data.py =="
echo "main $(git -C $APP rev-parse --short origin/main): $(cd $M && python -m pytest -q -p no:cacheprovider --collect-only tests/test_evals_data.py | tail -1)"
echo "head $(git -C $APP rev-parse --short $REV): $(cd $H && python -m pytest -q -p no:cacheprovider --collect-only tests/test_evals_data.py | tail -1)"
echo "== result summary: head tests x main evaluate_answers.py =="
(cd $B && python -m pytest -q -p no:cacheprovider tests/test_evals_data.py > out.txt 2>&1; tail -1 out.txt)
echo "== failing tests grouped by test function =="
grep '^FAILED' $B/out.txt | sed -E 's/^FAILED tests\/test_evals_data.py::([a-z_0-9]+).*/\1/' | sort | uniq -c
echo "== failing tests that are NOT new in this PR (must be empty) =="
git -C $APP diff origin/main...$REV -- tests/test_evals_data.py | grep -oE '^\+def (test_[a-z_0-9]+)' | sed 's/^+def //' | sort -u > $B/new.txt
grep '^FAILED' $B/out.txt | sed -E 's/^FAILED tests\/test_evals_data.py::([a-z_0-9]+).*/\1/' | sort -u | comm -23 - $B/new.txt
echo "(end)"
echo "== exception types in failures =="
grep -oE '^E +[A-Za-z]+(Error|Exception)' $B/out.txt | sort | uniq -c
