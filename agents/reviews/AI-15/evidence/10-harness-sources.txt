def defs(src):
    return {n.name: ast.dump(n, include_attributes=False) for n in ast.parse(src).body if isinstance(n, (ast.FunctionDef, ast.ClassDef))}

for kind in ("CHANGED", "added", "removed", "unchanged"):

"""QA AI-15 r1 mutation runner. Usage: mutants.py <app_tree> <test_file_source: head|main>
For each mutant: git-archive a fresh copy of the PR head into a new mktemp dir, apply the edit to
services/embedder/api.py (exact-match, must occur once), optionally swap in main's tests/test_verify.py,
run pytest -rA on tests/test_verify.py under the qa_stubs plugin, and report per-test outcomes + wall time.
TORCH_ONLY tests fail under MagicMock torch on head and main alike and are excluded from the verdict."""

HARNESS = os.path.dirname(os.path.abspath(__file__))
TORCH_ONLY = {"test_verify_pairs_takes_max_over_windows_and_keeps_order", "test_real_nli_separates_support_from_topicality"}

        return VerifyResponse(scores=scores, model=NLI_MODEL)

def wrap(block):
    lines = block.splitlines(keepends=True)
    head, body = lines[0], lines[1:]
    ind = re.match(r"\s*", body[0]).group(0)
    return head + ind + "with _ingest_lock:\n" + "".join("    " + l for l in body)


def swap(ce, fb):
    ce_l, fb_l = ce.splitlines(keepends=True), fb.splitlines(keepends=True)
    new_fb_head = fb_l[0].replace("elif", "if", 1)
    new_ce_head = ce_l[0].replace("if", "elif", 1)
    return new_fb_head + "".join(fb_l[1:]) + new_ce_head + "".join(ce_l[1:])


MUTANTS = {
    "m1 (AC5) lock the cross-encoder branch": [(CE, wrap(CE))],
    "m2 (AC5) lock only the bi-encoder fallback": [(FB, wrap(FB))],
    "m3 (AC5) lock the try: body of verify_endpoint": [(VT, wrap(VT))],
    "m4 (AC5) swap rerank branches (fallback first)": [(CE + FB, swap(CE, FB))],
    "q5 (QA) verify refuses 503 while an ingest holds the lock": [(VT, '''    if _ingest_lock.locked():
        raise HTTPException(status_code=503, detail="busy ingesting")
''' + VT)],
    "q6 (QA) rerank fallback locks only the passage embed": [(
        "            p_vecs = embedder.embed_texts(req.passages, is_query=False)\n",
        "            with _ingest_lock:\n                p_vecs = embedder.embed_texts(req.passages, is_query=False)\n")],
}


def run(name, edits):
    d = tempfile.mkdtemp(prefix="mut-", dir=os.path.join(os.path.dirname(HARNESS), "mutant-copies"))
    subprocess.run(f"git -C {APP} archive HEAD | tar -x -C {d}", shell=True, check=True)
    if SRC == "main":
        subprocess.run(f"git -C {APP} show origin/main:tests/test_verify.py > {d}/tests/test_verify.py", shell=True, check=True)
    p = os.path.join(d, "services/embedder/api.py")
    orig = open(p).read()
    s = orig
    for old, new in edits:
        if s.count(old) != 1:
            return name, "INVALID (anchor not found once)", [], 0.0, ""
        s = s.replace(old, new)
    open(p, "w").write(s)
    diff = "".join(l for l in difflib.unified_diff(orig.splitlines(True), s.splitlines(True), n=0)
                   if l[:1] in "+-" and not l.startswith(("+++", "---")))
    t0 = time.monotonic()
    r = subprocess.run([sys.executable, "-m", "pytest", "-p", "qa_stubs", "-p", "no:cacheprovider", "-q", "-rA",
                        "tests/test_verify.py"], cwd=d, capture_output=True, text=True, timeout=120,
                       env={**os.environ, "PYTHONPATH": f"{HARNESS}:{d}"})
    wall = time.monotonic() - t0
    fails = re.findall(r"^FAILED tests/test_verify.py::(\S+)", r.stdout, re.M)
    passed = re.findall(r"^PASSED tests/test_verify.py::(\S+)", r.stdout, re.M)
    errs = re.findall(r"^ERROR tests/test_verify.py::(\S+)", r.stdout, re.M)
    real = [f for f in fails + errs if f not in TORCH_ONLY]
    msgs = []
    for f in real:
        sec = re.search(r"_+ " + re.escape(f) + r" _+\n(.*?)(?=\n_+ test_|\n=+ )", r.stdout, re.S)
        el = [l[1:].strip() for l in (sec.group(1).splitlines() if sec else [])
              if l.startswith("E ") and ("!=" in l or "did not return" in l or "Error" in l)]
        msgs.append(f"{f}: " + (" | ".join(el[:3])[:240] if el else "?"))
    status = "KILLED" if real else "SURVIVED"
    return name, f"{status}  passed={len(passed)} failed(non-torch)={len(real)} torch-only-fails={len([f for f in fails if f in TORCH_ONLY])} wall={wall:.1f}s", msgs, wall, diff


results = [run(n, e) for n, e in (MUTANTS.items() if True else [])]
control = None
print(f"== SUMMARY (tests from {SRC}, api.py mutated in a git-archive copy of {subprocess.run(['git','-C',APP,'rev-parse','--short','HEAD'],capture_output=True,text=True).stdout.strip()}) ==")
for name, verdict, msgs, wall, diff in results:

for name, verdict, msgs, wall, diff in results:
    print(f"--- {name}\n{diff}")

"""QA AI-15 r1 pytest plugin: stand-ins for modules the sandbox cannot install (torch, transformers, fastapi,
psycopg, ulid, ...). fastapi is a minimal functional stub (decorators return the function unchanged, HTTPException
is a real Exception); everything else unimportable becomes a MagicMock module. Logic check only, not a torch run."""
import importlib.abc
import importlib.machinery
import importlib.util
import sys
import types
import uuid
from unittest.mock import MagicMock

# fastapi: functional minimal stub
fastapi = types.ModuleType("fastapi")


class HTTPException(Exception):
    def __init__(self, status_code, detail=None):
        super().__init__(status_code, detail)
        self.status_code, self.detail = status_code, detail


class FastAPI:
    def __init__(self, *a, **k):
        self.router = types.SimpleNamespace(on_startup=[])

    def _deco(self, *a, **k):
        return lambda f: f

    get = post = put = delete = _deco

    def on_event(self, kind):
        def reg(f):
            if kind == "startup":
                self.router.on_startup.append(f)
            return f
        return reg


fastapi.FastAPI, fastapi.HTTPException = FastAPI, HTTPException
sys.modules["fastapi"] = fastapi

# ulid: ulid.new() -> str-able unique id
ulid = types.ModuleType("ulid")
ulid.new = lambda: uuid.uuid4().hex
sys.modules["ulid"] = ulid

STUB_ROOTS = {"torch", "transformers", "psycopg", "confluent_kafka", "fitz", "pymupdf", "minio", "boto3",
              "tiktoken", "sentence_transformers", "uvicorn", "psycopg_pool", "pgvector", "tokenizers"}


class _Loader(importlib.abc.Loader):
    def create_module(self, spec):
        m = MagicMock(name=spec.name)
        m.__name__, m.__spec__, m.__path__ = spec.name, spec, []
        return m

    def exec_module(self, module):
        pass


class _Finder(importlib.abc.MetaPathFinder):
    def find_spec(self, name, path, target=None):
        root = name.split(".")[0]
        if root not in STUB_ROOTS:
            return None
        if root == name and importlib.machinery.PathFinder.find_spec(name) is not None:
            return None  # really installed: use it
        return importlib.util.spec_from_loader(name, _Loader(), is_package=True)


sys.meta_path.insert(0, _Finder())

#!/bin/bash
# QA AI-15 r1: run the 4 *not_serialized* tests 20x while N+2 busy-loop processes load the sandbox's N cores (timing-window flakiness).
set -u
APP=$1; H=$(dirname "$(readlink -f "$0")")
N=$(nproc); L=$((N + 2)); pids=()
for _ in $(seq $L); do python3 -c 'while True: pass' & pids+=($!); done
echo "cores=$N busy_loops=$L commit=$(git -C "$APP" rev-parse --short HEAD)"
pass=0; fail=0
for i in $(seq 20); do
  out=$(cd "$APP" && PYTHONPATH="$H:$APP" python3 -m pytest -p qa_stubs -p no:cacheprovider -q tests/test_verify.py -k not_serialized 2>&1 | tail -1)
  echo "run $i: $out"; case "$out" in *"4 passed"*) pass=$((pass+1));; *) fail=$((fail+1));; esac
done
kill "${pids[@]}" 2>/dev/null; wait 2>/dev/null
echo "TOTAL: $pass/20 runs with 4 passed, $fail runs otherwise"
