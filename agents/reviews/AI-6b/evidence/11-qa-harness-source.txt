	"fmt"
	"testing"
)

func TestQAProbeAI6b(t *testing.T) {
	const v = `{"claims": [{"text": "A.", "citations": [1]}]}`
	show := func(name, raw string, k int) {
		r, err := parseSynthesis(raw, k)
		fmt.Printf("PROBE %-46s claims=%v insufficient=%v droppedMarkers=%d droppedClaims=%d err=%v\n",
			name+":", r.Claims, r.InsufficientEvidence, r.DroppedMarkers, r.DroppedClaims, err)
	}
	show("P1 open fence, JSON, 3 trailing non-fence chars", "```json\n"+v+"xyz", 3)
	show("P2 open fence, JSON, two closing backticks", "```json\n"+v+"\n``", 3)
	show("P3 uppercase ```JSON fence", "```JSON\n"+v+"\n```", 3)
	show("P4 citations [1.0]", `{"claims": [{"text": "A.", "citations": [1.0]}]}`, 3)
	show("P5 capitalised keys Claims/Text/Citations", `{"Claims": [{"Text": "A.", "Citations": [1]}]}`, 3)

	show("P7 inline marker overflow [99999999999999999999]", `{"claims": [{"text": "A [99999999999999999999].", "citations": [1]}]}`, 3)
