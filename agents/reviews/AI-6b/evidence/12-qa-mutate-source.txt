def run(cmd, cwd):
    return subprocess.run(cmd, cwd=cwd, capture_output=True, text=True,
                          env={**os.environ, "GOTOOLCHAIN": "local", "GOFLAGS": "-mod=mod"})

PROBE = sys.argv[2] if len(sys.argv) > 2 and sys.argv[1] == "--probe" else None

def main():
    src = os.path.join(os.getcwd(), "internal/query")
    rows = []
    for mid, ac, desc, f, old, new in M:
        d = tempfile.mkdtemp(prefix=f"mut-{mid}-")
        shutil.copytree(MOD, d, dirs_exist_ok=True)
        for g in FILES:
            shutil.copy(os.path.join(src, g), os.path.join(d, "internal/query", g))
        p = os.path.join(d, "internal/query", f)
        body = open(p).read()
        if body.count(old) != 1:
            rows.append((mid, ac, desc, f"INVALID (pattern matched {body.count(old)}x)", "")); continue
        open(p, "w").write(body.replace(old, new))
        b = run(["go", "vet", "./internal/query/"], d)
        if b.returncode != 0:
            rows.append((mid, ac, desc, "INVALID (build/vet failed)", b.stderr.strip().splitlines()[-1][:90])); continue
        if PROBE:  # --probe <id>: run the QA probe test on this mutant and print its PROBE lines
            if mid != PROBE:
                continue
            shutil.copy(os.path.join(os.path.dirname(os.path.abspath(__file__)), "qa_probe_test.go"), os.path.join(d, "internal/query"))
            t = run(["go", "test", "-count=1", "-run", "TestQAProbeAI6b", "-v", "./internal/query/"], d)
            print(f"# probe on mutant {mid} ({desc}): {old!r} -> {new!r}")
            print("\n".join(l for l in t.stdout.splitlines() if l.startswith(("PROBE P1 ", "PROBE P2 "))))
            return
        t = run(["go", "test", "-count=1", "./internal/query/"], d)
        out = t.stdout + t.stderr
        failed = sorted(set(re.findall(r"^--- FAIL: (\w+)", out, re.M)))
        if "panic:" in out and not failed:
            failed = ["panic"]
        rows.append((mid, ac, desc, "KILLED" if t.returncode != 0 else "SURVIVED", ", ".join(failed)))
    k = sum(r[3] == "KILLED" for r in rows); s = sum(r[3] == "SURVIVED" for r in rows)
    print(f"SUMMARY: {len(rows)} mutants, {k} KILLED, {s} SURVIVED, {len(rows)-k-s} INVALID")
    for r in rows:
        print(f"{r[0]:<4} {r[1]:<6} {r[3]:<9} {r[2]}  ->  {r[4]}")

if __name__ == "__main__":
    main()
