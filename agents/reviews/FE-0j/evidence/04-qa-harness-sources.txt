    catch (e) { results.push({ name: t.name, ok: false, err: String(e.message || e).split('\n')[0] }); }
    cleanup(); }
  return results;
}

function matchers(v, neg) {
  const chk = (cond, msg) => { if (neg ? cond : !cond) fail((neg ? 'NOT ' : '') + msg); };
  return {
    toBe: (e) => chk(Object.is(v, e), `expected ${String(v)} toBe ${String(e)}`),
    toBeNull: () => chk(v === null, `expected ${v && v.outerHTML ? v.tagName : v} toBeNull`),
