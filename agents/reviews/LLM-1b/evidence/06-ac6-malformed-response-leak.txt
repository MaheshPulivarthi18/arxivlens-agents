func rawServer(t *testing.T, respond func(body, auth string) string) string {
	ln, err := net.Listen("tcp", "127.0.0.1:0")
	if err != nil {
		t.Fatal(err)
	}
	t.Cleanup(func() { ln.Close() })
	go func() {
		for {
			conn, err := ln.Accept()
			if err != nil {
				return
			}
			go func(c net.Conn) {
				defer c.Close()
				req, err := http.ReadRequest(bufio.NewReader(c))
				if err != nil {
					return
				}
				b, _ := io.ReadAll(req.Body)
				_, _ = io.WriteString(c, respond(string(b), req.Header.Get("Authorization")))
			}(conn)
		}
	}()
	return "http://" + ln.Addr().String()
}
