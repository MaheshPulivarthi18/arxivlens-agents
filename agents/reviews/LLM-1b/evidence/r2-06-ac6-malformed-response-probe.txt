// QA probe r2 (not part of the PR): round-1 probe 06 rerun, plus extra rows.
// A hostile raw-TCP server answers with malformed HTTP built from the request.
// Checks err.Error(), %+v/%#v and every error in the Unwrap chain.

	"bufio"

func qaRawServer(t *testing.T, respond func(body, auth string) string) string {
	ln, err := net.Listen("tcp", "127.0.0.1:0")
	if err != nil {
		t.Fatal(err)
	}
	t.Cleanup(func() { ln.Close() })
	go func() {
		for {
			conn, err := ln.Accept()
			if err != nil {
				return
			}
			go func(c net.Conn) {
				defer c.Close()
				req, err := http.ReadRequest(bufio.NewReader(c))
				if err != nil {
					return
				}
				b, _ := io.ReadAll(req.Body)
				_, _ = io.WriteString(c, respond(string(b), req.Header.Get("Authorization")))
			}(conn)
		}
	}()
	return "http://" + ln.Addr().String()
}

func TestQAProbeMalformedResponseLeak(t *testing.T) {
	const prompt = "SENTINEL-PROMPT"
	keys := map[string]string{"plain": "sk-SENTINEL-KEY-0123456789", "escapable": "sk-SENT\"INEL\\KEY"}
	bearer := func(a string) string { return strings.TrimPrefix(a, "Bearer ") }
	cases := []struct {
		name    string
		respond func(body, auth string) string
	}{
		{"r1: status line is the request body", func(b, a string) string { return b[:60] + "\r\n\r\n" }},
		{"r1: bad status code echoes prompt", func(b, a string) string { return "HTTP/1.1 " + prompt + " OK\r\n\r\n" }},
		{"r1: bad status code echoes key", func(b, a string) string { return "HTTP/1.1 " + bearer(a) + " OK\r\n\r\n" }},
		{"r1: bad header line echoes prompt", func(b, a string) string { return "HTTP/1.1 200 OK\r\n" + prompt + "\r\n\r\n" }},
		{"r2: bad header line echoes key", func(b, a string) string { return "HTTP/1.1 200 OK\r\n" + bearer(a) + "\r\n\r\n" }},
		{"r2: bad Content-Length echoes prompt", func(b, a string) string {
			return "HTTP/1.1 200 OK\r\nContent-Length: " + prompt + "\r\n\r\n"
		}},
		{"r2: bad Transfer-Encoding echoes prompt+key", func(b, a string) string {
			return "HTTP/1.1 200 OK\r\nTransfer-Encoding: " + prompt + bearer(a) + "\r\n\r\n"
		}},
		{"r2: bad chunk size echoes prompt (body read)", func(b, a string) string {
			return "HTTP/1.1 200 OK\r\nTransfer-Encoding: chunked\r\n\r\n" + prompt + "\r\n" + b + "\r\n"
		}},
		{"r2: bad chunked 503 echoes prompt (discard path)", func(b, a string) string {
			return "HTTP/1.1 503 x\r\nTransfer-Encoding: chunked\r\n\r\n" + prompt + bearer(a) + "\r\n"
		}},
	}
	for kn, key := range keys {
		for _, tt := range cases {
			t.Run(kn+"/"+tt.name, func(t *testing.T) {
				url := qaRawServer(t, tt.respond)
				c, err := New(Config{BaseURL: url, Model: "m", APIKey: NewSecret(key), Timeout: time.Second})
				if err != nil {
					t.Fatal(err)
				}
				_, err = c.Complete(context.Background(), Request{Messages: []Message{{Role: RoleUser, Content: prompt}}})
				outs := []string{err.Error(), fmt.Sprintf("%+v", err), fmt.Sprintf("%#v", err)}
				for e := errors.Unwrap(err); e != nil; e = errors.Unwrap(e) {
					outs = append(outs, e.Error(), fmt.Sprintf("%#v", e))
				}
				all := strings.Join(outs, " | ")
				keyBits := []string{key, fmt.Sprintf("%q", key), strings.Trim(fmt.Sprintf("%q", key), `"`)}
				leakK := false
				for _, k := range keyBits {
					leakK = leakK || strings.Contains(all, k)
				}
				leakP := strings.Contains(all, prompt) || strings.Contains(all, "SENTINEL")
				t.Logf("err=%q", err.Error())
				t.Logf("prompt leaked=%t key leaked=%t", leakP, leakK)
				if leakP || leakK {
					t.Errorf("LEAK in %q", all)
				}
			})
		}
	}
}
