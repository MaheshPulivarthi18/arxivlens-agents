// In-page WCAG contrast of an element's real computed text colour over the composited backgrounds of its ancestors.
const CONTRAST_FN = `(el) => {
  const parse = (s) => { let m = s.match(/rgba?\\(([^)]+)\\)/); if (m) { const p = m[1].split(/[ ,/]+/).filter(Boolean).map(Number); return [p[0], p[1], p[2], p[3] ?? 1]; }
    m = s.match(/color\\(srgb ([^)]+)\\)/); if (m) { const p = m[1].split(/[ /]+/).filter(Boolean).map(Number); return [p[0]*255, p[1]*255, p[2]*255, p[3] ?? 1]; }
    m = s.match(/oklch|oklab|lab|hsl/); if (m) { const c = document.createElement('canvas').getContext('2d'); c.fillStyle = s; c.fillRect(0,0,1,1); const d = c.getImageData(0,0,1,1).data; return [d[0], d[1], d[2], d[3]/255]; }
    return [0,0,0,0]; };
  const stack = []; for (let n = el; n; n = n.parentElement) stack.push(parse(getComputedStyle(n).backgroundColor));
  let bg = [255,255,255]; const root = parse(getComputedStyle(document.documentElement).backgroundColor); if (root[3] > 0) bg = root.slice(0,3);
  for (const c of stack.reverse()) bg = bg.map((v, i) => c[i] * c[3] + v * (1 - c[3]));
  const f = parse(getComputedStyle(el).color); const fg = f.slice(0,3).map((v, i) => v * f[3] + bg[i] * (1 - f[3]));
  const L = (rgb) => { const [r,g,b] = rgb.map((c) => { c /= 255; return c <= 0.04045 ? c/12.92 : ((c+0.055)/1.055)**2.4; }); return 0.2126*r + 0.7152*g + 0.0722*b; };
  const [a, b] = [L(fg), L(bg)].sort((x, y) => y - x); return Math.round(((a + 0.05) / (b + 0.05)) * 100) / 100; }`;

    toBeNull: () => ok(v === null, `toBeNull: got ${fmt(v)}`),
